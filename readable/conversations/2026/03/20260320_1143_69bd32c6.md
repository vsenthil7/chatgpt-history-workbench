# Minimal Python skeleton to start with

```python
from datetime import datetime
from typing import Dict, Any, List


def generate_structured_output(raw_input: str) -> Dict[str, Any]:
    """
    V1 placeholder.
    Later this can be replaced by an LLM call.
    For now, return mocked structured output.
    """
    return {
        "incident_id": "INC-001",
        "source": "email",
        "summary": "Customer data was shared with an external vendor without approval.",
        "severity": "high",
        "contains_pii": True,
        "external_sharing": True,
        "approved_vendor": False,
        "business_unit": "operations",
        "reported_by": "user",
        "recommended_action": "investigate"
    }


def validate_schema(data: Dict[str, Any]) -> Dict[str, Any]:
    required_fields = {
        "incident_id": str,
        "source": str,
        "summary": str,
        "severity": str,
        "contains_pii": bool,
        "external_sharing": bool,
        "approved_vendor": bool,
        "business_unit": str,
        "reported_by": str,
        "recommended_action": str
    }

    errors: List[str] = []

    for field, expected_type in required_fields.items():
        if field not in data:
            errors.append(f"Missing field: {field}")
        elif not isinstance(data[field], expected_type):
            errors.append(
                f"Invalid type for {field}: expected {expected_type.__name__}, got {type(data[field]).__name__}"
            )

    return {
        "valid": len(errors) == 0,
        "errors": errors
    }


def apply_policy_rules(data: Dict[str, Any]) -> Dict[str, Any]:
    triggered_rules = []

    if data.get("contains_pii") and data.get("external_sharing"):
        triggered_rules.append("RULE_1_PII_EXTERNAL_SHARING_ESCALATE")

    if (data.get("approved_vendor") is False) and data.get("external_sharing"):
        triggered_rules.append("RULE_2_UNAPPROVED_VENDOR_BLOCK_ESCALATE")

    if data.get("severity") == "high":
        triggered_rules.append("RULE_3_HIGH_SEVERITY_MANAGER_REVIEW")

    return {
        "triggered_rules": triggered_rules
    }


def decide_escalation(validation_result: Dict[str, Any], policy_result: Dict[str, Any]) -> Dict[str, Any]:
    if not validation_result["valid"]:
        return {
            "final_status": "validation_failed",
            "escalation": "block",
            "reason": "Schema validation failed"
        }

    rules = policy_result["triggered_rules"]

    if "RULE_2_UNAPPROVED_VENDOR_BLOCK_ESCALATE" in rules:
        return {
            "final_status": "blocked",
            "escalation": "escalate",
            "reason": "Unapproved external sharing detected"
        }

    if "RULE_1_PII_EXTERNAL_SHARING_ESCALATE" in rules:
        return {
            "final_status": "warning",
            "escalation": "escalate",
            "reason": "PII shared externally"
        }

    return {
        "final_status": "allowed",
        "escalation": "none",
        "reason": "No blocking policy triggered"
    }


def generate_audit(
    raw_input: str,
    structured_output: Dict[str, Any],
    validation_result: Dict[str, Any],
    policy_result: Dict[str, Any],
    escalation_result: Dict[str, Any]
) -> Dict[str, Any]:
    return {
        "timestamp": datetime.utcnow().isoformat() + "Z",
        "raw_input": raw_input,
        "structured_output": structured_output,
        "validation_result": validation_result,
        "policy_result": policy_result,
        "escalation_result": escalation_result
    }


def print_final_result(audit: Dict[str, Any]) -> None:
    print("\n=== FINAL RESULT ===")
    print(f"Incident ID: {audit['structured_output'].get('incident_id')}")
    print(f"Summary: {audit['structured_output'].get('summary')}")
    print(f"Validation Passed: {audit['validation_result']['valid']}")
    print(f"Triggered Rules: {audit['policy_result']['triggered_rules']}")
    print(f"Final Status: {audit['escalation_result']['final_status']}")
    print(f"Escalation: {audit['escalation_result']['escalation']}")
    print(f"Reason: {audit['escalation_result']['reason']}")
    print(f"Audit Timestamp: {audit['timestamp']}")


def main() -> None:
    raw_input = """
    A customer file containing personal information was sent by email
    to an external vendor who was not approved.
    """

    structured_output = generate_structured_output(raw_input)
    validation_result = validate_schema(structured_output)
    policy_result = apply_policy_rules(structured_output)
    escalation_result = decide_escalation(validation_result, policy_result)
    audit = generate_audit(
        raw_input,
        structured_output,
        validation_result,
        policy_result,
        escalation_result
    )
    print_final_result(audit)


if __name__ == "__main__":
    main()
```
