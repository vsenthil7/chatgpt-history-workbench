## 🧾 How Amazon actually handles GST (India)

### At sale time:
- Amazon collects:
  - Product price
  - GST from customer
- Amazon **deducts fees + GST on fees**
- Amazon remits **TCS (Tax Collected at Source)** to govt

---

### Example Settlement (simplified)

| Item | Amount (₹) |
|----|----|
| Gross sales (10 units × ₹699) | 6,990 |
| Amazon fees | -1,890 |
| GST on Amazon fees | -340 |
| Returns adjustment | -350 |
| TCS deducted | -70 |
| **Settlement credited to bank** | **₹4,340** |

Seller panic:
> “Sales ₹6,990, but I got only ₹4,340 😨”

Reality:
- GST input credit is **not lost**
- It is **delayed + fragmented across reports**

---

## 🔥 The Nightmare Reports (India-specific)

You must reconcile **4 files**, monthly:

1. Settlement report  
2. GST sales report  
3. TCS report (Form 26QG)  
4. Returns & reimbursements report  

Amazon does **not** merge these.

```javascript
function ingestLatestCSV(folderName, targetSheetName) {
  const folders = DriveApp.getFoldersByName(folderName);
  if (!folders.hasNext()) throw new Error("Folder not found: " + folderName);

  const folder = folders.next();
  const files = folder.getFilesByType(MimeType.CSV);
  let latestFile = null;

  while (files.hasNext()) {
    const file = files.next();
    if (!latestFile || file.getLastUpdated() > latestFile.getLastUpdated()) {
      latestFile = file;
    }
  }

  if (!latestFile) return;

  const csv = Utilities.parseCsv(latestFile.getBlob().getDataAsString());
  const sheet = SpreadsheetApp.getActive()
    .getSheetByName(targetSheetName);

  sheet.clearContents();
  sheet.getRange(1,1,csv.length,csv[0].length).setValues(csv);
  sheet.getRange(1,csv[0].length+1).setValue("load_ts");
  sheet.getRange(2,csv[0].length+1,csv.length-1,1)
       .setValue(new Date());
}
```

```javascript
function buildCoreFormulas_() {
  const ue = sh_("UNIT_ECONOMICS");
  // Copy Cost_Price from SKU_MASTER
  ue.getRange("H2").setFormula('=IF(A2="","",IFERROR(VLOOKUP(A2,SKU_MASTER!A:F,6,FALSE),""))');
  ue.getRange("H2:H200").autoFill(ue.getRange("H2:H200"), SpreadsheetApp.AutoFillSeries.DEFAULT_SERIES);

  // Referral fee amount
  ue.getRange("D2").setFormula('=IF(A2="","",B2*C2)');
  ue.getRange("D2:D200").autoFill(ue.getRange("D2:D200"), SpreadsheetApp.AutoFillSeries.DEFAULT_SERIES);

  // Total cost (Referral + FBA + Ads + Returns + Cost)
  ue.getRange("I2").setFormula('=IF(A2="","",SUM(D2,E2,F2,G2,H2))');
  ue.getRange("I2:I200").autoFill(ue.getRange("I2:I200"), SpreadsheetApp.AutoFillSeries.DEFAULT_SERIES);

  // Net profit
  ue.getRange("J2").setFormula('=IF(A2="","",B2-I2)');
  ue.getRange("J2:J200").autoFill(ue.getRange("J2:J200"), SpreadsheetApp.AutoFillSeries.DEFAULT_SERIES);

  // Profit status
  ue.getRange("K2").setFormula(`=IF(A2="","",IF(J2>=${CFG.minProfit},"SAFE","RISK"))`);
  ue.getRange("K2:K200").autoFill(ue.getRange("K2:K200"), SpreadsheetApp.AutoFillSeries.DEFAULT_SERIES);

  // Conditional formatting (simple)
  const rules = ue.getConditionalFormatRules();
  const netRange = ue.getRange("J2:J200");
  const safeRule = SpreadsheetApp.newConditionalFormatRule()
    .whenNumberGreaterThanOrEqualTo(CFG.minProfit).setBackground("#C6EFCE").setRanges([netRange]).build();
  const riskRule = SpreadsheetApp.newConditionalFormatRule()
    .whenNumberLessThan(CFG.minProfit).setBackground("#FFC7CE").setRanges([netRange]).build();
  ue.setConditionalFormatRules([safeRule, riskRule, ...rules]);

  // MODEL_ALERTS formulas (reads from UNIT_ECONOMICS + simulated sales/returns/inventory)
  const ma = sh_("MODEL_ALERTS");

  // Net_Profit + Profit_Status from UNIT_ECONOMICS
  ma.getRange("B2").setFormula('=IF(A2="","",IFERROR(VLOOKUP(A2,UNIT_ECONOMICS!A:K,10,FALSE),""))');
  ma.getRange("C2").setFormula('=IF(A2="","",IFERROR(VLOOKUP(A2,UNIT_ECONOMICS!A:K,11,FALSE),""))');

  // Units_Sold_4w from CLEAN_SETTLEMENTS (expects columns: Date, SKU_Code, Units, Gross)
  // If not present yet, simulation fills CLEAN_SETTLEMENTS.
  ma.getRange("D2").setFormula('=IF(A2="","",IFERROR(SUMIFS(CLEAN_SETTLEMENTS!C:C,CLEAN_SETTLEMENTS!B:B,A2),0))');

  // Returns_4w from CLEAN_RETURNS
  ma.getRange("E2").setFormula('=IF(A2="","",IFERROR(COUNTIF(CLEAN_RETURNS!B:B,A2),0))');

  // Return rate
  ma.getRange("F2").setFormula('=IF(A2="","",IF(D2=0,0,E2/D2))');

  // Units_Available + Avg_Daily_Sales from CLEAN_INVENTORY (expects columns: SKU_Code, Units_Available, Avg_Daily_Sales)
  ma.getRange("G2").setFormula('=IF(A2="","",IFERROR(VLOOKUP(A2,CLEAN_INVENTORY!A:C,2,FALSE),0))');
  ma.getRange("H2").setFormula('=IF(A2="","",IFERROR(VLOOKUP(A2,CLEAN_INVENTORY!A:C,3,FALSE),0))');

  // Days of cover
  ma.getRange("I2").setFormula('=IF(A2="","",IF(H2=0,"NA",G2/H2))');

  // Age bucket from days of cover (proxy; later you can use FC ageing report)
  ma.getRange("J2").setFormula('=IF(A2="","",IF(I2="NA","NA",IF(I2<=30,"0–30",IF(I2<=60,"30–60","60+"))))');

  // Cash freeze risk
  ma.getRange("K2").setFormula('=IF(A2="","",IF(OR(J2="60+",I2>60), "YES","NO"))');

  // Reimbursement due (placeholder rule: refunds not reimbursed for SUSPECTED_FRAUD)
  ma.getRange("L2").setFormula('=IF(A2="","",IF(COUNTIFS(CLEAN_RETURNS!B:B,A2,CLEAN_RETURNS!G:G,"SUSPECTED_FRAUD")>0,"CHECK",""))');

  // Scale eligibility (profit SAFE, return rate <= 0.15, days cover 10..30, no cash freeze)
  ma.getRange("M2").setFormula('=IF(A2="","",IF(AND(C2="SAFE",F2<=0.15,IF(I2="NA",FALSE,AND(I2>=10,I2<=30)),K2="NO"),"YES","NO"))');

  // Decision (SCALE/HOLD/STOP) — enterprise gate
  ma.getRange("N2").setFormula(`=IF(A2="","",
    IF(C2="RISK","STOP",
      IF(K2="YES","HOLD",
        IF(M2="YES","SCALE","HOLD")
      )
    )
  )`);

  // Autofill down
  ma.getRange("B2:N2").autoFill(ma.getRange("B2:N200"), SpreadsheetApp.AutoFillSeries.DEFAULT_SERIES);
}

function ingestLatestCsvFromFolder_(folder, targetSheetName) {
  const files = folder.getFilesByType(MimeType.CSV);
  let latest = null;
  while (files.hasNext()) {
    const f = files.next();
    if (!latest || f.getLastUpdated() > latest.getLastUpdated()) latest = f;
  }
  if (!latest) return;

  const csv = Utilities.parseCsv(latest.getBlob().getDataAsString());
  const sh = sh_(targetSheetName);
  sh.clearContents();
  sh.getRange(1,1,csv.length,csv[0].length).setValues(csv);
  sh.getRange(1,csv[0].length+1).setValue("load_ts");
  if (csv.length > 1) sh.getRange(2,csv[0].length+1,csv.length-1,1).setValue(new Date());
}

/** Prompt + strict schema for return clustering */
function callGeminiReturnCluster_(apiKey, reasonText) {
  const url = "https://generativelanguage.googleapis.com/v1beta/models/gemini-1.5-flash:generateContent?key=" + encodeURIComponent(apiKey);

  const prompt =
`Classify this Amazon return reason into EXACTLY one bucket.
Return reason text: "${reasonText}"

Buckets (choose one):
SIZE_FIT
DAMAGED
CUSTOMER_REMORSE
WRONG_ITEM
SUSPECTED_FRAUD
QUALITY
OTHER

Rules:
- Output MUST be JSON only.
- JSON schema: {"cluster":"<BUCKET>","confidence":0.0-1.0}
- No extra keys, no prose.`;

  const payload = {
    contents: [{ parts: [{ text: prompt }]}],
    generationConfig: { temperature: 0.1 }
  };

  const resp = UrlFetchApp.fetch(url, {
    method: "post",
    contentType: "application/json",
    payload: JSON.stringify(payload),
    muteHttpExceptions: true
  });

  const txt = resp.getContentText();
  try {
    const parsed = JSON.parse(txt);
    const outText = parsed.candidates?.[0]?.content?.parts?.[0]?.text || '{"cluster":"OTHER","confidence":0.3}';
    const j = JSON.parse(outText);
    return {
      cluster: (j.cluster || "OTHER").toString().trim(),
      confidence: Math.max(0, Math.min(1, Number(j.confidence || 0.3)))
    };
  } catch (e) {
    return { cluster: "OTHER", confidence: 0.3 };
  }
}
```
