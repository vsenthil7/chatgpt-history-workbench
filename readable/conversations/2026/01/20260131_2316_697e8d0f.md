```python
from pydantic import BaseModel, Field

from ibm_watsonx_orchestrate.flow_builder.flows import Flow, flow, START, END
from ibm_watsonx_orchestrate.flow_builder.types import Assignment, UserFieldKind
from ibm_watsonx_orchestrate.flow_builder.data_map import DataMap

# ----------------------------
# 1) Contracts (mirror your repo schema)
# ----------------------------
class IncidentInput(BaseModel):
    incident_id: str
    severity: str
    service_name: str
    summary: str
    signal_json: str
    runbook_key: str
    runbook_text: str

class PlanStep(BaseModel):
    step_id: str
    action: Literal["scale", "rollback", "restart", "verify", "notify"]
    target: str
    parameters: dict = Field(default_factory=dict)

class AIPlanV1(BaseModel):
    schema_version: Literal["1.0"]
    plan_id: str
    incident_id: str
    steps: List[PlanStep]

class PolicyDecision(BaseModel):
    decision: Literal["ALLOW", "REQUIRE_APPROVAL", "REJECT"]
    reason: str

class ApprovalResult(BaseModel):
    approval_status: Literal["APPROVED", "DENIED"]
    approver: str
    approval_comment: Optional[str] = ""

class FlowOutput(BaseModel):
    incident_id: str
    plan_json: dict
    decision: str
    approval_status: Optional[str] = None

# ----------------------------
# 2) Flow
# ----------------------------
@flow(
    name="h0010_p08_ibm_runtime",
    display_name="H0010 P08 IBM Runtime",
    description="IBM-only runtime: watsonx.ai plan -> policy -> HITL -> execution -> watsonx.data audit",
    input_schema=IncidentInput,
    output_schema=FlowOutput,
)
def build_flow(aflow: Flow = None) -> Flow:
    # --- GenAI node (Granite) with output_schema enforcement ---
    gen = aflow.prompt(
        name="gen_plan",
        display_name="Generate Plan (Granite JSON)",
        system_prompt=[
            "You are a runbook planner. Output MUST be valid JSON that conforms exactly to the provided schema.",
            "Rules: include schema_version='1.0'; no additional properties; whitelisted actions only.",
            "Return JSON only."
        ],
        user_prompt=[
            "Incident:\n{flow.input}\n\nRunbook:\n{flow.input.runbook_text}\n\nProduce plan JSON."
        ],
        # pick your Granite model available in your tenant
        llm="ibm/granite-3-3-8b-instruct",
        output_schema=AIPlanV1,   # <-- the contract enforcement lever
    )

    # --- Policy decision as HITL-safe default (you will replace with deterministic kernel tool call) ---
    # For IBM-only runtime: implement your policy kernel as an Orchestrate Tool (Python tool or OpenAPI tool)
    # and call it here. For now: require approval for P1/P2 by rule in mapping (replace later).
    # We'll collect decision via a user activity (acts as the policy "gate" until tool is plugged in).

    policy_gate = aflow.userflow()
    decision_node = policy_gate.field(
        direction="input",
        name="decision",
        display_name="Policy decision",
        kind=UserFieldKind.Choice,
        text="Policy result for this plan?",
        option={"labels": ["ALLOW", "REQUIRE_APPROVAL", "REJECT"], "values": ["ALLOW", "REQUIRE_APPROVAL", "REJECT"]},
    )
    reason_node = policy_gate.field(
        direction="input",
        name="reason",
        display_name="Reason",
        kind=UserFieldKind.Text,
        text="Reason (short)",
    )
    policy_gate.edge(START, decision_node)
    policy_gate.edge(decision_node, reason_node)
    policy_gate.edge(reason_node, END)

    # --- Approval userflow (HITL) ---
    approval_flow = aflow.userflow()
    approve_node = approval_flow.field(
        direction="input",
        name="approval_status",
        display_name="Approve execution?",
        kind=UserFieldKind.Boolean,
        text="Approve running the plan steps now?",
    )
    approver_node = approval_flow.field(
        direction="input",
        name="approver",
        display_name="Approver",
        kind=UserFieldKind.Text,
        text="Your name / handle",
    )
    comment_node = approval_flow.field(
        direction="input",
        name="approval_comment",
        display_name="Comment",
        kind=UserFieldKind.Text,
        text="Optional comment",
    )
    approval_flow.edge(START, approve_node)
    approval_flow.edge(approve_node, approver_node)
    approval_flow.edge(approver_node, comment_node)
    approval_flow.edge(comment_node, END)

    # --- Wire the main path ---
    aflow.edge(START, gen)
    aflow.sequence(gen, policy_gate)

    # Branching: if REQUIRE_APPROVAL -> approval_flow else skip
    branch = aflow.branch(name="branch_on_policy")

    # Map policy decision out of userflow nodes
    dm_decision = DataMap()
    dm_decision.add(Assignment(
        target_variable="self.input.condition",
        value_expression='flow["userflow_1"]["Policy decision"].output.value == "REQUIRE_APPROVAL"'
    ))
    branch.input_map = dm_decision

    aflow.edge(policy_gate, branch)

    # If approval required, run approval userflow
    aflow.sequence(branch.true, approval_flow)

    # End: return output (execution + watsonx.data writeback nodes get plugged here next)
    aflow.edge(branch.false, END)
    aflow.edge(approval_flow, END)

    return aflow
```
