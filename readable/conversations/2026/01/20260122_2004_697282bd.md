import asyncio
import re
from pathlib import Path
from urllib.parse import urlparse

from playwright.async_api import async_playwright


def safe_name(s: str, max_len: int = 120) -> str:
    s = re.sub(r"[^\w\- ]+", "", s, flags=re.UNICODE).strip()
    s = re.sub(r"\s+", " ", s)
    if not s:
        s = "page"
    return s[:max_len].strip().replace(" ", "_")


async def auto_scroll(page, steps=18, pause_ms=350):
    # Notion loads blocks progressively; scrolling helps ensure full content appears
    for _ in range(steps):
        await page.mouse.wheel(0, 1600)
        await page.wait_for_timeout(pause_ms)


async def extract_visible_text(page) -> str:
    # Pull readable text from rendered page (Notion is JS-rendered)
    # We take body innerText; it’s the most robust “what you see is what you get”.

def normalize(url: str) -> str:
    url = url.split("#")[0]
    return url.rstrip("/")


def safe_name(s: str, max_len: int = 120) -> str:
    s = re.sub(r"[^\w\- ]+", "", s).strip()
    s = re.sub(r"\s+", " ", s)
    return (s[:max_len] or "page").replace(" ", "_")


async def auto_scroll(page, steps=20):
    for _ in range(steps):
        await page.mouse.wheel(0, 1800)
        await page.wait_for_timeout(300)


async def extract_links(page):
    return await page.evaluate("""
        () => Array.from(document.querySelectorAll('a[href]'))
            .map(a => a.href)
            .filter(h => h.includes('notion.so'))
    """)


async def extract_text(page):
    title = await page.title()
    text = await page.evaluate("() => document.body.innerText")
    return title.strip(), text.strip()


async def run():
    OUT_DIR.mkdir(exist_ok=True)
    seen = set()
    queue = [normalize(ROOT_URL)]

    async with async_playwright() as p:
        browser = await p.chromium.launch(headless=True)
        page = await browser.new_page()

        while queue:
            url = normalize(queue.pop(0))
            if url in seen:
                continue
            seen.add(url)

            print(f"→ {url}")
            try:
                await page.goto(url, wait_until="networkidle", timeout=120_000)
                await auto_scroll(page)

                title, text = await extract_text(page)
