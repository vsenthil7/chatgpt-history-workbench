```python
import re
import sys
import json
from pathlib import Path
from urllib.parse import urlparse

from playwright.sync_api import sync_playwright, TimeoutError as PlaywrightTimeoutError


OUTPUT_DIR = Path("share_md_output")
OUTPUT_DIR.mkdir(exist_ok=True)


def slugify(text: str, max_len: int = 80) -> str:
    text = re.sub(r"[^\w\s-]", "", text, flags=re.UNICODE)
    text = re.sub(r"[\s_-]+", "-", text).strip("-")
    return (text[:max_len] or "untitled").lower()


def detect_source(url: str) -> str:
    host = urlparse(url).netloc.lower()
    if "claude.ai" in host:
        return "claude"
    if "chatgpt.com" in host or "openai.com" in host:
        return "chatgpt"
    return "unknown"


def clean_text(text: str) -> str:
    text = text.replace("\r\n", "\n")
    text = re.sub(r"\n{3,}", "\n\n", text)
    return text.strip()


def extract_title(page, source: str) -> str:
    try:
        title = page.title().strip()
        if title:
            return title
    except Exception:
        pass

    selectors = [
        "h1",
        "header h1",
        "[data-testid='conversation-title']",
        "[class*='title']",
    ]

    for sel in selectors:
        try:
            el = page.locator(sel).first
            if el.count() > 0:
                txt = el.inner_text(timeout=2000).strip()
                if txt:
                    return txt
        except Exception:
            pass

    return f"{source}-share"


def extract_chatgpt(page) -> str:
    chunks = []

    candidate_selectors = [
        "main",
        "article",
        "[role='main']",
        "body",
    ]

    for sel in candidate_selectors:
        try:
            loc = page.locator(sel).first
            if loc.count() > 0:
                txt = loc.inner_text(timeout=4000).strip()
                if txt and len(txt) > 200:
                    chunks.append(txt)
        except Exception:
            pass

    if not chunks:
        return ""

    best = max(chunks, key=len)
    return clean_text(best)


def extract_claude(page) -> str:
    chunks = []

    candidate_selectors = [
        "main",
        "[role='main']",
        "article",
        "body",
    ]

    for sel in candidate_selectors:
        try:
            loc = page.locator(sel).first
            if loc.count() > 0:
                txt = loc.inner_text(timeout=4000).strip()
                if txt and len(txt) > 200:
                    chunks.append(txt)
        except Exception:
            pass

    if not chunks:
        return ""

    best = max(chunks, key=len)
    return clean_text(best)


def extract_content(page, source: str) -> str:
    if source == "chatgpt":
        return extract_chatgpt(page)
    if source == "claude":
        return extract_claude(page)

    try:
        return clean_text(page.locator("body").inner_text(timeout=4000))
    except Exception:
        return ""


def to_markdown(title: str, url: str, source: str, body: str) -> str:
    return f"""# {title}

- **Source:** {source}
- **URL:** {url}

---

```text
{body}
```
"""


def process_url(browser, url: str) -> dict:
    source = detect_source(url)
    page = browser.new_page()

    try:
        page.goto(url, wait_until="networkidle", timeout=60000)
    except PlaywrightTimeoutError:
        try:
            page.goto(url, wait_until="domcontentloaded", timeout=60000)
        except Exception as e:
            return {"url": url, "ok": False, "error": f"Navigation failed: {e}"}
    except Exception as e:
        return {"url": url, "ok": False, "error": f"Navigation failed: {e}"}

    try:
        page.wait_for_timeout(5000)
        title = extract_title(page, source)
        body = extract_content(page, source)

        if not body or len(body) < 100:
            return {
                "url": url,
                "ok": False,
                "error": "Could not extract enough visible text. The page may require login, extra rendering, or changed DOM structure.",
            }

        filename = OUTPUT_DIR / f"{slugify(title)}.md"
        filename.write_text(to_markdown(title, url, source, body), encoding="utf-8")

        return {
            "url": url,
            "ok": True,
            "title": title,
            "file": str(filename),
            "chars": len(body),
        }
    finally:
        page.close()


def main():
    if len(sys.argv) > 1:
        urls = sys.argv[1:]
    else:
        input_file = Path("share_links.txt")
        if not input_file.exists():
            print("Usage:")
            print("  python share_to_md.py <url1> <url2> ...")
            print("or create share_links.txt with one URL per line.")
            sys.exit(1)
        urls = [line.strip() for line in input_file.read_text(encoding="utf-8").splitlines() if line.strip()]

    results = []

    with sync_playwright() as p:
        browser = p.chromium.launch(headless=True)
        try:
            for url in urls:
                print(f"Processing: {url}")
                result = process_url(browser, url)
                results.append(result)
                print(json.dumps(result, ensure_ascii=False, indent=2))
        finally:
            browser.close()

    summary_file = OUTPUT_DIR / "summary.json"
    summary_file.write_text(json.dumps(results, ensure_ascii=False, indent=2), encoding="utf-8")
    print(f"\nSaved summary: {summary_file}")


if __name__ == "__main__":
    main()
```
