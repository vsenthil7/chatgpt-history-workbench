## Core Script (V1 Working)

```python
from pdf2image import convert_from_path
import pytesseract
import pandas as pd
import re

# Step 1: Convert PDF to images
images = convert_from_path("tirunelveli_roll.pdf", dpi=300)

data = []

for i, img in enumerate(images):
    text = pytesseract.image_to_string(img, lang='eng+tam')

    lines = text.split("\n")

    for line in lines:
        # Basic voter pattern (tune later)
        match = re.search(r'(\d+)\s+([A-Za-z\s]+)\s+(\d+)\s+(Male|Female)', line)
        if match:
            data.append({
                "serial_no": match.group(1),
                "name": match.group(2),
                "age": match.group(3),
                "gender": match.group(4),
                "booth_page": i
            })

df = pd.DataFrame(data)
df.to_csv("tirunelveli_voters_raw.csv", index=False)

print("Done: Raw extraction ready")
```
