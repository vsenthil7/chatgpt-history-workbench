luckynumber.move
module ctf::luckynumber {
    use ctf::counter::{Self, Counter};

    fun init(ctx: &mut TxContext) {
        counter::create_counter(ctx);
    }

    public struct Flag has key, store {
        id: UID,
        user: address
    }

    public entry fun get_flag(user_counter: &mut Counter, lucky_num: u64, ctx: &mut TxContext) {
        counter::increment(user_counter);
        counter::is_within_limit(user_counter);

        let _ = lucky_num;
        transfer::public_transfer(Flag {
            id: object::new(ctx),
            user: tx_context::sender(ctx)
        }, tx_context::sender(ctx));
    }
}


counter.move
module ctf::counter {

    const MaxCounter: u64 = 1000;
    const ENoAttemptLeft: u64 = 0;
    
    /// A shared counter.
    public struct Counter has key {
        id: UID,
        owner: address,
        value: u64
    }

    /// Create and share a Counter object.
    public(package) fun create_counter(ctx: &mut TxContext) {
        transfer::share_object(Counter {
            id: object::new(ctx),
            owner: tx_context::sender(ctx),
            value: 0
        })
    }

    public fun owner(counter: &Counter): address {
        counter.owner
    }

    public fun value(counter: &Counter): u64 {
        counter.value
    }

    /// Increment a counter by 1.
    public fun increment(counter: &mut Counter) {
        counter.value = counter.value + 1;
    }

    /// Set value (only runnable by the Counter owner)
    public entry fun set_value(counter: &mut Counter, value: u64, ctx: &TxContext) {
        assert!(counter.owner == tx_context::sender(ctx), 0);
        counter.value = value;
    }

    /// Check whether the counter has reached the limit.
    public fun is_within_limit(counter: &mut Counter) {
        assert!(counter.value <= MaxCounter, ENoAttemptLeft);
    }
}

### What the code really means (so you don’t overthink “lucky_num”)
In `get_flag`, the `lucky_num` is **assigned to `_` and never used**:

```move
let _ = lucky_num;
```

So **any u64 works**. The only gating condition is the shared `Counter` staying `<= 1000` after increment. There’s **no owner check** inside `increment()` / `is_within_limit()`, so **anyone can call it** until the counter exceeds 1000. citeturn0search1turn2view1

## Solve steps (CLI)
**Estimated completion time:** 10–15 minutes.

### 0) Ensure you’re on the same network as the CTF (Testnet)

```bash
iota client active-env
```

### 2) (Optional but good) Inspect the shared Counter object
Counter ID given:
`0x03730ebe89151b43e696c7c6ca6e32818ececf12459d8b67a7173dfce361ffa5`


You’re checking it exists and is the expected `ctf::counter::Counter` and seeing current `value`.

### 3) Call `get_flag` (any lucky number)
Package:
`0xc8bc4a1780bcdfec4ca4a85522c6a2d9f2dad8ce6881fe6bb5d6df21214ffc69`

```bash
iota client call \
  --package 0xc8bc4a1780bcdfec4ca4a85522c6a2d9f2dad8ce6881fe6bb5d6df21214ffc69 \
  --module luckynumber \
  --function get_flag \
  --args 0x03730ebe89151b43e696c7c6ca6e32818ececf12459d8b67a7173dfce361ffa5 7 \
  --gas-budget 10000000
```

Notes:
- `--args` are **ordered exactly like the function signature**. citeturn2view1  
- `7` can be **any** u64 (0, 1, 777, 1337…).

### 4) Find your Flag object ID (to paste into the CTF “Flag Id” box)
After the call succeeds, a `Flag` object is transferred to your sender address. List your owned objects:

```bash
iota client objects
```

On Windows PowerShell, to quickly spot it:
```powershell
iota client objects | Select-String -Pattern "Flag|luckynumber|ctf"
```

Then confirm the object:
```bash
iota client object <FLAG_OBJECT_ID>
```

Copy that `<FLAG_OBJECT_ID>` into the website’s **“Enter Flag Object Id”**.

### If it fails with “no attempt left”
That means the shared counter’s `value` is already **> 1000** (or becomes >1000 after your increment). In that case, the challenge is effectively “burned” on that shared object and you’ll need the CTF to provide/reset a new counter object ID (or use whatever new ID they publish).
