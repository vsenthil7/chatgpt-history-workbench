```diff
+from __future__ import annotations
+
+from dataclasses import dataclass
+from enum import Enum
+
+
+class SessionState(str, Enum):
+    CREATED = "created"
+    CONSENTED = "consented"
+    ACTIVE = "active"
+    CLOSED = "closed"
+    ESCALATED = "escalated"
+
+
+@dataclass(frozen=True)
+class InvalidTransition(Exception):
+    from_state: SessionState
+    to_state: SessionState
+
+    def __str__(self) -> str:
+        return f"Invalid transition: {self.from_state} -> {self.to_state}"
+
+
+ALLOWED = {
+    SessionState.CREATED: {SessionState.CONSENTED},
+    SessionState.CONSENTED: {SessionState.ACTIVE, SessionState.ESCALATED},
+    SessionState.ACTIVE: {SessionState.CLOSED, SessionState.ESCALATED},
+    SessionState.CLOSED: set(),
+    SessionState.ESCALATED: set(),
+}
+
+
+def transition(current: SessionState, target: SessionState) -> SessionState:
+    if target in ALLOWED[current]:
+        return target
+    raise InvalidTransition(current, target)
```

```diff
+from __future__ import annotations
+
+from dataclasses import dataclass
+from enum import Enum
+from typing import List
+
+
+class RiskFlag(str, Enum):
+    SELF_HARM = "self_harm"
+    VIOLENCE = "violence"
+    ABUSE = "abuse"
+    CRISIS = "crisis"
+
+
+@dataclass(frozen=True)
+class RiskResult:
+    escalate: bool
+    flags: List[RiskFlag]
+    reason: str  # deterministic explanation logged and stored
+
+
+# Deterministic phrase triggers (keep minimal + auditable; extend carefully with governance)
+SELF_HARM_TRIGGERS = [
+    "kill myself",
+    "end my life",
+    "suicide",
+    "self harm",
+]
+VIOLENCE_TRIGGERS = [
+    "kill them",
+    "shoot",
+    "stab",
+    "hurt someone",
+]
+ABUSE_TRIGGERS = [
+    "i was abused",
+    "sexual abuse",
+    "domestic violence",
+]
+CRISIS_TRIGGERS = [
+    "i can't go on",
+    "no reason to live",
+    "emergency",
+]
+
+
+def _match_any(text: str, phrases: List[str]) -> bool:
+    t = text.lower()
+    return any(p in t for p in phrases)
+
+
+def assess_risk(user_text: str) -> RiskResult:
+    flags: List[RiskFlag] = []
+
+    if _match_any(user_text, SELF_HARM_TRIGGERS):
+        flags.append(RiskFlag.SELF_HARM)
+    if _match_any(user_text, VIOLENCE_TRIGGERS):
+        flags.append(RiskFlag.VIOLENCE)
+    if _match_any(user_text, ABUSE_TRIGGERS):
+        flags.append(RiskFlag.ABUSE)
+    if _match_any(user_text, CRISIS_TRIGGERS):
+        flags.append(RiskFlag.CRISIS)
+
+    if flags:
+        return RiskResult(
+            escalate=True,
+            flags=flags,
+            reason=f"Deterministic trigger match: {', '.join([f.value for f in flags])}",
+        )
+
+    return RiskResult(escalate=False, flags=[], reason="No deterministic triggers matched")
```

```diff
+
+    def add_message(self, session_id: str, role: str, content: str, ts: datetime) -> None:
+        msg = MessageRecord(session_id=session_id, role=role, content=content, ts=ts)
+        self.db.add(msg)
+        self.db.commit()
+
+    def list_messages(self, session_id: str) -> Iterable[MessageRecord]:
+        stmt = select(MessageRecord).where(MessageRecord.session_id == session_id).order_by(MessageRecord.id)
+        return list(self.db.execute(stmt).scalars().all())
+
+    def mark_escalated(
+        self,
+        rec: SessionRecord,
+        *,
+        flags_csv: str,
+        reason: str,
+        handoff_packet: dict,
+        now: datetime,
+    ) -> None:
+        rec.escalated = True
+        rec.escalation_flags_csv = flags_csv
+        rec.escalation_reason = reason
+        rec.handoff_packet_json = json.dumps(handoff_packet, ensure_ascii=False, default=str)
+        rec.updated_at = now
+        self.db.commit()
```

## 14) API routes — `src/reflectsafe/api/v1/routes_sessions.py`

```diff
+++ b/src/reflectsafe/api/v1/routes_sessions.py
@@
+from __future__ import annotations
+
+import logging
+import uuid
+from datetime import datetime, timezone
+
+from fastapi import APIRouter, Depends, HTTPException, status
+from sqlalchemy.orm import Session
+
+from reflectsafe.domain.handoff import generate_handoff_packet
+from reflectsafe.domain.prompt_engine import build_reflection_prompt
+from reflectsafe.domain.risk_rules import assess_risk
+from reflectsafe.domain.state_machine import SessionState, transition
+from reflectsafe.storage.db import SessionLocal
+from reflectsafe.storage.repo import SessionRepo
+
+from .schemas import (
+    CloseResponse,
+    ConsentRequest,
+    HandoffPacketResponse,
+    MessageRequest,
+    MessageResponse,
+    SessionCreateResponse,
+    SessionStateResponse,
+)
+
+
+log = logging.getLogger("reflectsafe.api.v1.sessions")
+router = APIRouter(prefix="/v1/sessions", tags=["sessions"])
+
+
+def now_utc() -> datetime:
+    return datetime.now(timezone.utc).replace(tzinfo=None)  # store naive UTC for portability
+
+
+def get_db() -> Session:
+    db = SessionLocal()
+    try:
+        yield db
+    finally:
+        db.close()
+
+
+@router.post("", response_model=SessionCreateResponse, status_code=status.HTTP_201_CREATED)
+def create_session(db: Session = Depends(get_db)) -> SessionCreateResponse:
+    repo = SessionRepo(db)
+    sid = uuid.uuid4().hex
+    now = now_utc()
+    rec = repo.create_session(sid, now)
+    return SessionCreateResponse(session_id=rec.id, state=rec.state, created_at=rec.created_at)
+
+
+@router.post("/{session_id}/consent", response_model=SessionStateResponse)
+def capture_consent(session_id: str, req: ConsentRequest, db: Session = Depends(get_db)) -> SessionStateResponse:
+    repo = SessionRepo(db)
+    rec = repo.get_session(session_id)
+    if not rec:
+        raise HTTPException(status_code=404, detail="session_not_found")
+
+    current = SessionState(rec.state)
+    try:
+        new_state = transition(current, SessionState.CONSENTED)
+    except Exception:
+        raise HTTPException(status_code=409, detail="invalid_state_transition")
+
+    now = now_utc()
+    repo.capture_consent(rec, actor=req.actor, consent_text=req.consent_text, now=now)
+    repo.update_state(rec, new_state, now)
+    log.info("consent_captured", extra={"session_id": session_id, "actor": req.actor})
+    return SessionStateResponse(
+        session_id=rec.id, state=rec.state, consent_captured=rec.consent_captured, escalated=rec.escalated
+    )
+
+
+@router.post("/{session_id}/messages", response_model=MessageResponse)
+def post_message(session_id: str, req: MessageRequest, db: Session = Depends(get_db)) -> MessageResponse:
+    repo = SessionRepo(db)
+    rec = repo.get_session(session_id)
+    if not rec:
+        raise HTTPException(status_code=404, detail="session_not_found")
+
+    if rec.state in (SessionState.CLOSED.value, SessionState.ESCALATED.value) or rec.escalated:
+        raise HTTPException(status_code=423, detail="session_locked")
+
+    if not rec.consent_captured:
+        raise HTTPException(status_code=412, detail="consent_required")
+
+    # Ensure ACTIVE state once message begins (after consent)
+    current = SessionState(rec.state)
+    if current == SessionState.CONSENTED:
+        try:
+            repo.update_state(rec, transition(current, SessionState.ACTIVE), now_utc())
+        except Exception:
+            raise HTTPException(status_code=409, detail="invalid_state_transition")
+
+    # Deterministic risk assessment first
+    risk = assess_risk(req.user_text)
+    repo.add_message(session_id, role="user", content=req.user_text, ts=now_utc())
+    rec.last_user_text = req.user_text
+    db.commit()
+
+    if risk.escalate:
+        # Transition to escalated + generate handoff packet + lock
+        created_at = rec.created_at
+        escalated_at = now_utc()
+        msgs = repo.list_messages(session_id)
+        timeline = [f"{m.ts.isoformat()}Z {m.role}: {m.content[:200]}" for m in msgs]
+        excerpt = req.user_text
+        summary = "Escalation triggered by deterministic risk rules. Human follow-up required."
+
+        packet = generate_handoff_packet(
+            session_id=session_id,
+            created_at=created_at,
+            escalated_at=escalated_at,
+            risk_flags=risk.flags,
+            risk_reason=risk.reason,
+            user_words_excerpt=excerpt,
+            timeline=timeline,
+            summary_for_human=summary,
+            consent_captured=rec.consent_captured,
+            consent_actor=rec.consent_actor,
+            consent_at=rec.consent_at,
+        )
+
+        # Lock state
+        repo.update_state(rec, transition(SessionState(rec.state), SessionState.ESCALATED), escalated_at)
+        repo.mark_escalated(
+            rec,
+            flags_csv=",".join([f.value for f in risk.flags]),
+            reason=risk.reason,
+            handoff_packet=packet.__dict__ | {"risk_flags": [f.value for f in packet.risk_flags]},
+            now=escalated_at,
+        )
+        log.info("session_escalated", extra={"session_id": session_id, "reason": risk.reason})
+
+        return MessageResponse(
+            session_id=session_id,
+            state=rec.state,
+            risk_escalated=True,
+            risk_reason=risk.reason,
+            flags=[f.value for f in risk.flags],
+            reflection_prompt=None,
+        )
+
+    # Non-escalated: return structured reflection prompt (execution by LLM optional later)
+    pr = build_reflection_prompt(req.user_text)
+    return MessageResponse(
+        session_id=session_id,
+        state=rec.state,
+        risk_escalated=False,
+        risk_reason=risk.reason,
+        flags=[],
+        reflection_prompt=pr.prompt,
+    )
+
+
+@router.post("/{session_id}/close", response_model=CloseResponse)
+def close_session(session_id: str, db: Session = Depends(get_db)) -> CloseResponse:
+    repo = SessionRepo(db)
+    rec = repo.get_session(session_id)
+    if not rec:
+        raise HTTPException(status_code=404, detail="session_not_found")
+    if rec.escalated or rec.state == SessionState.ESCALATED.value:
+        raise HTTPException(status_code=423, detail="session_locked_escalated")
+
+    now = now_utc()
+    try:
+        repo.update_state(rec, transition(SessionState(rec.state), SessionState.CLOSED), now)
+    except Exception:
+        raise HTTPException(status_code=409, detail="invalid_state_transition")
+    return CloseResponse(session_id=session_id, state=rec.state)
+
+
+@router.get("/{session_id}/handoff", response_model=HandoffPacketResponse)
+def get_handoff(session_id: str, db: Session = Depends(get_db)) -> HandoffPacketResponse:
+    repo = SessionRepo(db)
+    rec = repo.get_session(session_id)
+    if not rec:
+        raise HTTPException(status_code=404, detail="session_not_found")
+    if not rec.handoff_packet_json:
+        raise HTTPException(status_code=404, detail="handoff_not_available")
+    return HandoffPacketResponse(session_id=session_id, handoff_packet_json=rec.handoff_packet_json)
```
