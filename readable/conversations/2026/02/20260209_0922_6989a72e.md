import hashlib
from dataclasses import dataclass
from typing import Optional

@dataclass(frozen=True)
class HashChainRecord:
    payload_hash: str
    prev_hash: Optional[str]
    chain_hash: str

def sha256_hex(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()

def chain_hash(payload_bytes: bytes, prev_chain_hash: Optional[str]) -> HashChainRecord:
    payload_hash = sha256_hex(payload_bytes)
    base = (prev_chain_hash or "").encode() + payload_hash.encode()
    ch = sha256_hex(base)
    return HashChainRecord(payload_hash=payload_hash, prev_hash=prev_chain_hash, chain_hash=ch)
```

from dataclasses import dataclass
from typing import Dict, Any, Optional
import json
from opsguard.core.tamper import chain_hash, HashChainRecord

@dataclass(frozen=True)
class AuditEvent:
    event_type: str
    actor: str
    payload: Dict[str, Any]
    record: HashChainRecord

class AuditLog:
    def __init__(self) -> None:
        self._events: list[AuditEvent] = []
        self._last_chain_hash: Optional[str] = None

    def append(self, event_type: str, actor: str, payload: Dict[str, Any]) -> AuditEvent:
        payload_bytes = json.dumps(payload, sort_keys=True).encode()
        rec = chain_hash(payload_bytes, self._last_chain_hash)
        ev = AuditEvent(event_type=event_type, actor=actor, payload=payload, record=rec)
        self._events.append(ev)
        self._last_chain_hash = rec.chain_hash
        return ev

    def verify(self) -> bool:
        prev = None
        for ev in self._events:
            payload_bytes = json.dumps(ev.payload, sort_keys=True).encode()
            rec = chain_hash(payload_bytes, prev)
            if rec.chain_hash != ev.record.chain_hash:
                return False
            prev = ev.record.chain_hash
        return True
```
