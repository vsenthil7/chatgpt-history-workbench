## 1️⃣ Canon Registry (Runtime Component)

**File:** `canon/registry.py`

```python
import json
import hashlib
import os
from typing import Dict


class CanonIntegrityError(Exception):
    pass


class CanonRegistry:
    def __init__(self, canon_path: str, expected_hash: str):
        self.canon_path = canon_path
        self.expected_hash = expected_hash
        self._canon: Dict = {}

    def load(self) -> Dict:
        if not os.path.exists(self.canon_path):
            raise CanonIntegrityError("Canon file missing")

        with open(self.canon_path, "rb") as f:
            raw = f.read()

        actual_hash = hashlib.sha256(raw).hexdigest()
        if actual_hash != self.expected_hash:
            raise CanonIntegrityError(
                f"Canon hash mismatch: {actual_hash} != {self.expected_hash}"
            )

        self._canon = json.loads(raw.decode("utf-8"))
        return self._canon

    def is_loaded(self) -> bool:
        return bool(self._canon)
```
