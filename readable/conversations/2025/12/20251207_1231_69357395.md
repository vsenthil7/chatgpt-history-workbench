## 4️⃣ Real-time reflex logic (<70ms prediction path)

Targeted latency budget (edge device):

1. **Sensor → Raindrop (5–10ms)**
   - Read sensor buffer.
   - Compute basic features (acceleration, speed, etc.).
   - Pack to `RaindropEvent`.

2. **SmartMemory Update (5–15ms)**
   - Append to circular buffer (e.g. last 2s at 50Hz → 100 points).
   - Build minimal feature window (e.g. last 10–20 frames).

3. **SmartInference Model (15–25ms)**
   - Load pre-compiled model (ONNX / C++ runtime).
   - Run inference on small window (e.g. 20×N features).
   - Output: `event_type`, `risk_score`.

4. **Reflex Policy & Action Trigger (5–10ms)**
   - Evaluate:
     - if `risk_score > zone_threshold`
     - check cooldown + alert state
   - Trigger:
     - local buzzer/light
     - send message to voice agent
     - push alert to cloud.

Total target: **30–60ms** on modest edge hardware (e.g. ARM SBC), aiming <70ms even under load.

# 1️⃣ **RaindropEvent — core event object**

```python
from dataclasses import dataclass, field
from typing import Dict, Any
import time
import uuid

@dataclass
class RaindropEvent:
    timestamp: float
    entity_id: str
    event_type: str       # "IMU", "FALL_OBJ", "MOTION", "CAM_FEATURE"
    features: Dict[str, Any]
    event_id: str = field(default_factory=lambda: str(uuid.uuid4()))

    @staticmethod
    def make(entity_id: str, event_type: str, features: Dict[str, Any]):
        return RaindropEvent(
            timestamp=time.time(),
            entity_id=entity_id,
            event_type=event_type,
            features=features,
        )
```

# 2️⃣ **SmartMemory — sliding window context**

Create file:

`src/edge/smart_memory.py`

```python
from collections import deque
from typing import List
from raindrop_event import RaindropEvent

class SmartMemory:
    def __init__(self, window_seconds: float = 2.0, fps: int = 50):
        self.max_len = int(window_seconds * fps)
        self.buffer = deque(maxlen=self.max_len)

    def add(self, event: RaindropEvent):
        self.buffer.append(event)

    def get_window(self) -> List[RaindropEvent]:
        return list(self.buffer)

    def to_feature_window(self):
        """
        Converts raindrop events → numerical feature matrix.
        Example: last 20 frames × N features.
        """
        window = self.get_window()
        feat_list = []

        for e in window:
            row = []
```

```python
import numpy as np
from typing import Dict, Any

class SmartInference:
    def __init__(self, mode="rules", model_path=None):
        self.mode = mode
        self.model = None

        if mode == "onnx" and model_path is not None:
            import onnxruntime as ort
            self.model = ort.InferenceSession(model_path)

    def predict(self, feature_window: list) -> Dict[str, Any]:
        if not feature_window:
            return {"risk": 0.0, "event_type": "NONE", "confidence": 0.0}

        if self.mode == "rules":
            return self._rule_prediction(feature_window)
        else:
            return self._model_prediction(feature_window)

    def _rule_prediction(self, window):
        """
        Heuristic-based reflex:
        Detect slip: sudden spikes in horizontal acceleration
        Detect falling object: high downward velocity
        """
        arr = np.array(window)

        # Core features we expect
        ax = arr[:, 0] if arr.shape[1] > 0 else np.zeros(len(arr))
        ay = arr[:, 1] if arr.shape[1] > 1 else np.zeros(len(arr))
        az = arr[:, 2] if arr.shape[1] > 2 else np.zeros(len(arr))

        # Slip detector
        slip_risk = float(np.max(np.abs(ax)) + np.max(np.abs(ay)))
        slip_risk_norm = min(1.0, slip_risk / 15.0)

        # Falling object detector
        downward_vel = float(np.abs(np.min(az) - 9.8))
        fall_risk_norm = min(1.0, downward_vel / 10.0)

        # Final decision
        if slip_risk_norm > 0.6:
            return {"risk": slip_risk_norm, "event_type": "SLIP", "confidence": 0.85}

        if fall_risk_norm > 0.6:
            return {"risk": fall_risk_norm, "event_type": "FALLING_OBJECT", "confidence": 0.80}

        return {"risk": 0.1, "event_type": "NORMAL", "confidence": 0.5}

    def _model_prediction(self, window):
        import numpy as np
        x = np.array(window, dtype=np.float32)[None, :, :]

        inp = {self.model.get_inputs()[0].name: x}
        out = self.model.run(None, inp)[0][0]

        return {
            "risk": float(out[0]),
            "event_type": "MODEL",
            "confidence": float(out[1])
        }
```

### **Notes**
- Rule mode gives **instant accuracy** for slip + fall  
- ONNX mode allows future upgrade  
- Returns **risk, event_type, confidence** for reflex policy

```python
import time

class ReflexPolicy:
    def __init__(self, slip_threshold=0.6, fall_threshold=0.6, cooldown=1.0):
        self.slip_threshold = slip_threshold
        self.fall_threshold = fall_threshold
        self.cooldown = cooldown
        self.last_alert = 0

    def decide(self, prediction):
        now = time.time()
        if now - self.last_alert < self.cooldown:
            return None  # cooldown active

        event = prediction["event_type"]
        risk = prediction["risk"]

        if event == "SLIP" and risk > self.slip_threshold:
            self.last_alert = now
            return {
                "type": "SLIP_RISK_HIGH",
                "severity": "HIGH",
                "message": "Slip risk detected"
            }

        if event == "FALLING_OBJECT" and risk > self.fall_threshold:
            self.last_alert = now
            return {
                "type": "FALLING_OBJECT_HIGH",
                "severity": "HIGH",
                "message": "Object falling detected"
            }

        return None
```

```python
import time
from typing import Dict, Any

class ReflexPolicy:
    """
    Maps prediction → alert dict (or None), with cooldown.
    """

    def __init__(self, slip_threshold: float = 0.6,
                 fall_threshold: float = 0.6,
                 cooldown: float = 1.0):
        self.slip_threshold = slip_threshold
        self.fall_threshold = fall_threshold
        self.cooldown = cooldown
        self.last_alert = 0.0

    def decide(self, prediction: Dict[str, Any]) -> Dict[str, Any] | None:
        now = time.time()
        if now - self.last_alert < self.cooldown:
            return None

        event_type = prediction.get("event_type")
        risk = float(prediction.get("risk", 0.0))

        if event_type == "SLIP" and risk > self.slip_threshold:
            self.last_alert = now
            return {
                "type": "SLIP_RISK_HIGH",
                "severity": "HIGH",
                "message": "Slip risk detected",
            }

        if event_type == "FALLING_OBJECT" and risk > self.fall_threshold:
            self.last_alert = now
            return {
                "type": "FALLING_OBJECT_HIGH",
                "severity": "HIGH",
                "message": "Object falling detected",
```

```python
class SmartMemory:
```

```python
def add(self, event: RaindropEvent):
    self.buffer.append(event)
```

```python
def get_window(self) -> List[RaindropEvent]:
    return list(self.buffer)
```

```python
def to_feature_window(self):
    window = self.get_window()
    feat_list = []
    for e in window:
        row = []
        for _, v in e.features.items():
            row.append(float(v))
        feat_list.append(row)
    return feat_list
```

```python
def _model_prediction(self, window: List[List[float]]) -> Dict[str, Any]:
```
