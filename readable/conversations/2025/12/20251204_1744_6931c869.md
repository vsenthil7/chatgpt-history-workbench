## **file_loader.py**
```python
import pandas as pd
import os

def load_excels(folder_path):
    files = [f for f in os.listdir(folder_path) if f.endswith(".xlsx")]
    dataframes = []

    for file in files:
        df = pd.read_excel(os.path.join(folder_path, file))
        df["source_file"] = file
        dataframes.append(df)

    if not dataframes:
        raise ValueError("No Excel files found.")

    return pd.concat(dataframes, ignore_index=True)
```

```python
import pandas as pd
import os

def save_summary(df, output_folder="output"):
    os.makedirs(output_folder, exist_ok=True)
    summary = df.groupby("Product")["Sales"].sum().reset_index()
    output_path = os.path.join(output_folder, "summary_report.xlsx")
    summary.to_excel(output_path, index=False)
    print("✔ Summary report generated.")
```

## **gui_main.py (EXE Launcher GUI)**

```python
import tkinter as tk
from tkinter import filedialog, messagebox
from file_loader import load_excels
from cleaner import clean_data
from summary_report import save_summary
from chart_generator import make_chart

def run_process(folder):
    df = load_excels(folder)
    df = clean_data(df)
    save_summary(df)
    make_chart(df)

def select_folder():
    folder = filedialog.askdirectory()
    if folder:
        try:
            run_process(folder)
            messagebox.showinfo("Success", "Processing complete!")
        except Exception as e:
            messagebox.showerror("Error", str(e))

app = tk.Tk()
app.title("Excel Automation Engine")

button = tk.Button(app, text="Select Folder and Run", command=select_folder)
button.pack(pady=20, padx=20)

app.mainloop()
```
