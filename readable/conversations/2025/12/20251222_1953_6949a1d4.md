## IMPLEMENTATION — ERNIE CLIENT (REAL)

### `src/ernie/client.py`

```python
import os
import requests

class ErnieClient:
    def __init__(self):
        self.api_key = os.getenv("ERNIE_API_KEY")
        self.endpoint = os.getenv("ERNIE_API_ENDPOINT")

        if not self.api_key or not self.endpoint:
            raise RuntimeError("ERNIE credentials not set")

    def generate(self, prompt, temperature=0.2):
        payload = {
            "prompt": prompt,
            "temperature": temperature
        }

        headers = {
            "Authorization": f"Bearer {self.api_key}",
            "Content-Type": "application/json"
        }

        response = requests.post(
            self.endpoint,
            headers=headers,
            json=payload,
            timeout=30
        )

        response.raise_for_status()
        return response.json()
```

```python
# src/ernie/client.py
import requests, os

class ErnieClient:
    def __init__(self):
        self.key = os.getenv("ERNIE_API_KEY")
        self.url = os.getenv("ERNIE_API_ENDPOINT")

    def call(self, prompt):
        r = requests.post(
            self.url,
            headers={"Authorization": f"Bearer {self.key}"},
            json={"prompt": prompt}
        )
        r.raise_for_status()
        return r.json()["output"]
```

```python
from fastapi import FastAPI, UploadFile
from ingest.pdf_loader import load_pdf
from ocr.paddle_layout import extract_layout
from docmap.build_map import build_docmap
from decision.identity import infer_identity
from decision.outputs import decide_outputs

app = FastAPI()

@app.post("/process")
async def process_pdf(file: UploadFile):
    run_id, pdf = load_pdf(file)
    layout = extract_layout(pdf)
    docmap = build_docmap(layout)
    identity = infer_identity(docmap)
```
