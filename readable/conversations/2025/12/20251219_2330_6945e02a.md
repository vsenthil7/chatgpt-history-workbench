```jsx
import { useState } from "react";
import { reflect } from "../lib/ai";

export default function SafetyReflection({ onNext, onBack }) {
  const [input, setInput] = useState("");
  const [output, setOutput] = useState("");
  const [loading, setLoading] = useState(false);

  async function runReflection() {
    setLoading(true);
    const res = await reflect(input);
    setOutput(res.text);
    setLoading(false);
  }

  return (
    <>
      <h1 className="text-xl font-bold mb-4">Safety Reflection</h1>

      <textarea
        className="w-full border p-2 mb-4"
        placeholder="What stands out to you?"
        value={input}
        onChange={(e) => setInput(e.target.value)}
      />

      <button
        onClick={runReflection}
        className="mb-4 px-4 py-2 bg-gray-800 text-white"
      >
        {loading ? "Reflecting..." : "Generate Reflection"}
      </button>

      {output && (
        <div className="border p-3 mb-4 bg-gray-50">
          <strong>Reflection:</strong>
          <p className="mt-2">{output}</p>
        </div>
      )}

      <div className="flex justify-between">
        <button onClick={onBack}>Back</button>
        <button onClick={onNext} className="px-4 py-2 bg-black text-white">
          Continue
        </button>
      </div>
    </>
  );
}
```
