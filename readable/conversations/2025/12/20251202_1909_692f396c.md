dataframes = []

    if file.endswith(".xlsx"):
        df = pd.read_excel(f"{input_path}/{file}")
        df['SourceFile'] = file  # traceability
        dataframes.append(df)

# Merge all files
merged_df = pd.concat(dataframes, ignore_index=True)

# Clean: remove duplicates + sort
merged_df.drop_duplicates(inplace=True)
merged_df.sort_values(by=["Date", "Product"], inplace=True)

# Add total amount
merged_df["Total"] = merged_df["Quantity"] * merged_df["Price"]

# Save merged output
merged_output = f"{output_path}/merged_sales.xlsx"
merged_df.to_excel(merged_output, index=False)

print("✔ Merge complete.")
print(f"Output saved as: {merged_output}")
print("Rows:", len(merged_df))
