### `src/ocr/paddle_ocr.py`
```python
from paddleocr import PaddleOCR

class PaddleOCRVL:
    def __init__(self, lang="en"):
        self.ocr = PaddleOCR(use_angle_cls=True, lang=lang)

    def extract(self, file_path: str) -> dict:
        result = self.ocr.ocr(file_path, cls=True)
        return {
            "blocks": [
                {"text": line[1][0], "confidence": line[1][1]}
                for page in result for line in page
            ]
        }
```

import os
import requests

ERNIE_ENDPOINT = os.getenv("ERNIE_API_URL")
ERNIE_KEY = os.getenv("ERNIE_API_KEY")

class ErnieClient:
    def call(self, prompt: str) -> dict:
        headers = {"Authorization": f"Bearer {ERNIE_KEY}"}
        payload = {"prompt": prompt, "temperature": 0.2}
        r = requests.post(ERNIE_ENDPOINT, json=payload, headers=headers, timeout=60)
        r.raise_for_status()
        return r.json()

from ocr.paddle_ocr import PaddleOCRVL

def run_v1(file_path: str):
    ocr = PaddleOCRVL()
    ernie = ErnieClient()
    router = DocumentRouter()
    generator = ContentGenerator()
    validator = OutputValidator()

import os, requests

def get_access_token() -> str:
    api_key = os.getenv("QIANFAN_API_KEY")
    secret_key = os.getenv("QIANFAN_SECRET_KEY")
    if not api_key or not secret_key:
        raise RuntimeError("Missing QIANFAN_API_KEY / QIANFAN_SECRET_KEY in environment.")

    # Baidu token generation flow is referenced by Qianfan SDK docs (API key + secret key -> access token).
    # Exact token URL/params are defined in Baidu docs referenced by qianfan SDK.

class PaddleOCRVL:

class ErnieClient:
    def call(self, prompt: str) -> dict:

        payload = {"prompt": prompt, "temperature": 0.2}
        r = requests.post(ERNIE_ENDPOINT, json=payload, headers=headers, timeout=60)
