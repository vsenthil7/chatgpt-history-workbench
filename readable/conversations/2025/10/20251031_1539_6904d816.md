# Pattern Studio - Day 1
# Relationship: B = k*A , C = m*A

def compute_pattern(k, m, A=None, B=None, C=None):
    # Forward mode
    if A is not None:
        B_calc = k * A
        C_calc = m * A
        return B_calc, C_calc

    # Reverse check mode
    if B is not None and C is not None:
        A1 = B / k
        A2 = C / m
        consistent = abs(A1 - A2) < 1e-6
        return A1, A2, consistent

```python
# Relationship:  B = k*A ,  C = m*A

def validate_row(k, m, A=None, B=None, C=None, tol=1e-9):
    """
    Validate or infer proportional pattern values.
    Returns a dictionary with predictions and consistency check.
    """
    result = {"k": k, "m": m, "A_in": A, "B_in": B, "C_in": C}

    # --- Forward mode (if A given)
    if A is not None:
        result["B_pred"] = k * A
        result["C_pred"] = m * A

    # --- Reverse mode (if B or C given)
    if B is not None and k != 0:
        result["A_from_B"] = B / k
    if C is not None and m != 0:
        result["A_from_C"] = C / m

    # --- Consistency check
    if "A_from_B" in result and "A_from_C" in result:
        result["consistent"] = abs(result["A_from_B"] - result["A_from_C"]) <= tol
    elif "A_from_B" in result or "A_from_C" in result:
        result["consistent"] = True   # single-source inference assumed consistent
    else:
        result["consistent"] = None

    return result
```
