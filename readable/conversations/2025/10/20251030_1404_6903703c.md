# 1) Upload to GCS **with gzip on the fly** (fastest)
If your source files are local (or on the worker) and you just want them gzipped in GCS.

**BashOperator (gsutil does the compression):**
```python
from airflow import DAG
from airflow.operators.bash import BashOperator
from datetime import datetime

with DAG(
    "gzip_upload_to_gcs",
    start_date=datetime(2025, 10, 1),
    schedule_interval=None,
    catchup=False,
) as dag:
    # -m = parallel, -Z = gzip on upload (for compressible types)
    gzip_upload = BashOperator(
        task_id="gzip_upload",
        bash_command=(
            "gsutil -m cp -Z /path/to/*.csv gs://my-bucket/raw/"
        )
    )
```
Notes:
- `-Z` compresses *as it uploads* for texty types (csv, json, ndjson, txt, html, etc.).  
- GCS will store the object with `Content-Encoding: gzip` and serve it transparently.

---

# 2) **Compress existing GCS objects** (no server-side gzip; do it in a task)
If files are already in GCS, download → gzip → re-upload (optionally delete originals).

**PythonOperator with the Google provider:**
```python
import gzip, shutil, os, tempfile
from airflow import DAG
from airflow.operators.python import PythonOperator
from datetime import datetime
from airflow.providers.google.cloud.hooks.gcs import GCSHook

BUCKET = "my-bucket"
SRC_PREFIX = "raw/2025-10-30/"
DST_PREFIX = "raw/2025-10-30-gz/"

def gzip_gcs_prefix():
    hook = GCSHook()  # uses google_cloud_default by default
    blobs = hook.list(BUCKET, prefix=SRC_PREFIX)

    with tempfile.TemporaryDirectory() as tmp:
        for blob in blobs:
            if blob.endswith("/"):  # skip "folders"
                continue
            local_path = os.path.join(tmp, os.path.basename(blob))
            hook.download(bucket_name=BUCKET, object_name=blob, filename=local_path)

            gz_path = local_path + ".gz"
            with open(local_path, "rb") as f_in, gzip.open(gz_path, "wb") as f_out:
                shutil.copyfileobj(f_in, f_out)

            dst_object = DST_PREFIX + os.path.basename(blob) + ".gz"
            hook.upload(
                bucket_name=BUCKET,
                object_name=dst_object,
                filename=gz_path,
                mime_type="application/gzip",
            )

            # optional: remove the original
            # hook.delete(BUCKET, blob)

with DAG(
    "gzip_existing_gcs_objects",
    start_date=datetime(2025, 10, 1),
    schedule_interval=None,
    catchup=False,
) as dag:
    compress_task = PythonOperator(
        task_id="gzip_prefix",
        python_callable=gzip_gcs_prefix
    )
```
Tips:
- This works in Cloud Composer (managed Airflow) out of the box if you’ve set up the GCP connection/permissions.
- For big volumes, parallelize with a task-map (dynamic task mapping) over the `blobs` list.

# 3) **Bundle many files into one archive** (zip/tar.gz) and store in GCS
Good for archiving daily partitions, then move the single archive.

```python
import tarfile, os, tempfile
from airflow import DAG
from airflow.operators.python import PythonOperator
from datetime import datetime
from airflow.providers.google.cloud.hooks.gcs import GCSHook

BUCKET = "my-bucket"
SRC_PREFIX = "raw/2025-10-30/"
ARCHIVE_OBJECT = "archives/2025-10-30.tar.gz"

def tar_gz_prefix():
    hook = GCSHook()
    blobs = [b for b in hook.list(BUCKET, prefix=SRC_PREFIX) if not b.endswith("/")]

    with tempfile.TemporaryDirectory() as tmp:
        local_files = []
        for blob in blobs:
            lp = os.path.join(tmp, os.path.basename(blob))
            hook.download(BUCKET, blob, lp)
            local_files.append(lp)

        archive_path = os.path.join(tmp, "archive.tar.gz")
        with tarfile.open(archive_path, "w:gz") as tar:
            for f in local_files:
                tar.add(f, arcname=os.path.basename(f))

        hook.upload(BUCKET, ARCHIVE_OBJECT, archive_path, mime_type="application/gzip")

with DAG(
    "archive_partition_to_tar_gz",
    start_date=datetime(2025, 10, 1),
    schedule_interval=None,
    catchup=False,
) as dag:
    archive_task = PythonOperator(
        task_id="archive_prefix",
        python_callable=tar_gz_prefix
    )
```

## Practical choices
- **Fastest & cheapest** when possible: use `gsutil cp -Z` at upload time (Pattern #1).
- **Already in GCS and large sets**: consider dynamic task mapping to parallelize Pattern #2.
- **Serving to downstream systems**: keeping per-file `gzip` with correct `Content-Encoding` is better than a single tarball (which readers must unpack).
- **BigQuery external tables**: they can read `.gz` CSV/JSON directly—keep one file per shard/partition for parallelism.
