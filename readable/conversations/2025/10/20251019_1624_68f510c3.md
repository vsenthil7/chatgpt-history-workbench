```python
import os
import sys
import glob
import logging
import traceback
from typing import List, Optional, Dict, Any

import pandas as pd

try:
    import yaml
except ImportError:
    print("Missing dependency 'pyyaml'. Install with: pip install pyyaml")
    sys.exit(1)

try:
    import chardet
except ImportError:
    chardet = None  # optional; we’ll handle gracefully


def load_config(path: str) -> Dict[str, Any]:
    if not os.path.exists(path):
        raise FileNotFoundError(f"Config file not found: {path}")
    with open(path, "r", encoding="utf-8") as f:
        return yaml.safe_load(f)


def setup_logging(log_path: str):
    logging.basicConfig(
        filename=log_path,
        filemode="w",
        level=logging.INFO,
        format="%(asctime)s [%(levelname)s] %(message)s",
    )
    console = logging.StreamHandler(sys.stdout)
    console.setLevel(logging.INFO)
    console.setFormatter(logging.Formatter("%(message)s"))
    logging.getLogger().addHandler(console)


def detect_encoding(file_path: str, default: str = "utf-8") -> str:
    if not chardet:
        return default
    try:
        with open(file_path, "rb") as f:
            raw = f.read(50000)
            res = chardet.detect(raw)
            enc = res.get("encoding") or default
            return enc
    except Exception:
        return default


def read_csv_safely(file_path: str, delimiter: str, encoding_pref: str) -> pd.DataFrame:
    enc = detect_encoding(file_path) if encoding_pref == "auto" else encoding_pref
    try:
        return pd.read_csv(file_path, encoding=enc, sep=delimiter)
    except Exception as e:
        logging.warning(f"CSV read failed with {enc}: {e}. Retrying with utf-8.")
        return pd.read_csv(file_path, encoding="utf-8", sep=delimiter, engine="python")


def read_excel_sheets(file_path: str, sheet_name: Optional[str], header_row: Optional[int]) -> Dict[str, pd.DataFrame]:
    """
    Returns dict[sheet_name] -> DataFrame
    """
    try:
        if sheet_name and sheet_name != "*":
            df = pd.read_excel(file_path, sheet_name=sheet_name, header=header_row, engine="openpyxl")
            return {sheet_name: df}
        else:
            dfs = pd.read_excel(file_path, sheet_name=None, header=header_row, engine="openpyxl")
            # dfs is already dict of sheet -> df
            return dfs
    except Exception as e:
        logging.error(f"Failed reading Excel '{file_path}': {e}")
        return {}


def unify_columns(dfs: List[pd.DataFrame]) -> List[pd.DataFrame]:
    # union of all columns (as strings)
    all_cols = set()
    for df in dfs:
        all_cols.update(map(str, df.columns))
    all_cols = list(all_cols)
    unified = []
    for df in dfs:
        # ensure string column names
        df.columns = list(map(str, df.columns))
        unified.append(df.reindex(columns=all_cols))
    return unified


def merge_files(
    input_dir: str,
    include_patterns: List[str],
    sheet_name: Optional[str],
    header_row: Optional[int],
    add_source_columns: bool,
    deduplicate: bool,
    dedup_key_columns: List[str],
    csv_delimiter: str,
    encoding_pref: str,
) -> (pd.DataFrame, pd.DataFrame):
    records = []
    summary_rows = []

    for pattern in include_patterns:
        for path in glob.glob(os.path.join(input_dir, pattern)):
            if os.path.isdir(path):
                continue
            ext = os.path.splitext(path)[1].lower()

            if ext in [".csv", ".tsv"]:
                delim = csv_delimiter if ext == ".csv" else "\t"
                try:
                    df = read_csv_safely(path, delimiter=delim, encoding_pref=encoding_pref)
                    if add_source_columns:
                        df["source_file"] = os.path.basename(path)
                        df["source_sheet"] = ""
                    records.append(df)
                    summary_rows.append({
                        "source_file": os.path.basename(path),
                        "source_sheet": "",
                        "rows": len(df)
                    })
                    logging.info(f"Loaded CSV: {os.path.basename(path)} rows={len(df)}")
                except Exception as e:
                    logging.error(f"Error reading CSV '{path}': {e}\n{traceback.format_exc()}")

            elif ext in [".xlsx", ".xls"]:
                sheets = read_excel_sheets(path, sheet_name, header_row)
                for sname, df in sheets.items():
                    if not isinstance(df, pd.DataFrame):
                        continue
                    if add_source_columns:
                        df["source_file"] = os.path.basename(path)
                        df["source_sheet"] = sname
                    records.append(df)
                    summary_rows.append({
                        "source_file": os.path.basename(path),
                        "source_sheet": sname,
                        "rows": len(df)
                    })
                logging.info(f"Loaded Excel: {os.path.basename(path)} sheets={len(sheets)}")
            else:
                logging.info(f"Skipped unsupported file: {os.path.basename(path)}")

    if not records:
        logging.warning("No data files found to merge.")
        return pd.DataFrame(), pd.DataFrame()

    # Unify column sets, then concatenate
    unified = unify_columns(records)
    merged = pd.concat(unified, ignore_index=True)

    # Optional deduplication
    if deduplicate:
        if dedup_key_columns:
            before = len(merged)
            merged = merged.drop_duplicates(subset=dedup_key_columns)
            logging.info(f"Deduplicated by keys {dedup_key_columns}: {before} -> {len(merged)}")
        else:
            before = len(merged)
            merged = merged.drop_duplicates()
            logging.info(f"Deduplicated full rows: {before} -> {len(merged)}")

    summary = pd.DataFrame(summary_rows)
    if not summary.empty:
        summary = summary.groupby(["source_file", "source_sheet"], as_index=False)["rows"].sum()
        summary["total_rows_after_merge"] = len(merged)

    return merged, summary


def save_excel(merged: pd.DataFrame, summary: pd.DataFrame, out_path: str):
    with pd.ExcelWriter(out_path, engine="openpyxl") as writer:
        merged.to_excel(writer, index=False, sheet_name="Merged")
        if not summary.empty:
            summary.to_excel(writer, index=False, sheet_name="Summary")


def main():
    cfg_path = "config.yml"
    cfg = load_config(cfg_path)

    input_dir = cfg.get("input_dir", "./input")
    include_patterns = cfg.get("include_patterns", ["*.xlsx", "*.xls", "*.csv"])
    sheet_name = cfg.get("sheet_name", "*")
    header_row = cfg.get("header_row", 0)
    add_source_columns = bool(cfg.get("add_source_columns", True))
    deduplicate = bool(cfg.get("deduplicate", True))
    dedup_key_columns = cfg.get("dedup_key_columns", [])
    csv_delimiter = cfg.get("csv_delimiter", ",")
    encoding_pref = cfg.get("encoding", "auto")
    output_file = cfg.get("output_file", "merged.xlsx")
    log_file = cfg.get("log_file", "mergebot.log")

    os.makedirs(input_dir, exist_ok=True)
    setup_logging(log_file)

    logging.info("=== ExcelMergeBot starting ===")
    logging.info(f"Input dir: {os.path.abspath(input_dir)}")

    try:
        merged, summary = merge_files(
            input_dir=input_dir,
            include_patterns=include_patterns,
            sheet_name=sheet_name,
            header_row=header_row,
            add_source_columns=add_source_columns,
            deduplicate=deduplicate,
            dedup_key_columns=dedup_key_columns or [],
            csv_delimiter=csv_delimiter,
            encoding_pref=encoding_pref,
        )

        if merged.empty:
            logging.warning("Nothing merged. Check your input files and patterns.")
        else:
            save_excel(merged, summary, output_file)
            logging.info(f"Saved: {os.path.abspath(output_file)}")
            if not summary.empty:
                logging.info("Summary generated.")
    except Exception as e:
        logging.error(f"Fatal error: {e}\n{traceback.format_exc()}")

    logging.info("=== ExcelMergeBot done ===")


if __name__ == "__main__":
    main()
```

```python
def read_excel_sheets(file_path: str, sheet_name: Optional[str], header_row: Optional[int]) -> Dict[str, pd.DataFrame]:
    try:
        if sheet_name and sheet_name != "*":
            df = pd.read_excel(file_path, sheet_name=sheet_name, header=header_row, engine="openpyxl")
            return {sheet_name: df}
        else:
            dfs = pd.read_excel(file_path, sheet_name=None, header=header_row, engine="openpyxl")
            return dfs
    except Exception as e:
        logging.error(f"Failed reading Excel '{file_path}': {e}")
        return {}

def normalize_column_name(name: Any, rules: Dict[str, Any]) -> str:
    s = str(name)
    if rules.get("strip_spaces", True):
        s = s.strip()
    if rules.get("collapse_inner_spaces", True):
        s = re.sub(r"\s+", " ", s)
    if rules.get("lowercase", False):
        s = s.lower()
    return s


def apply_column_mappings(cols: List[str], mappings: List[Dict[str, Any]]) -> List[str]:
    out = []
    for c in cols:
        new_name = c
        for rule in mappings or []:
            pat = rule.get("match")
            to = rule.get("to", c)
            is_regex = bool(rule.get("regex", False))
            try:
                if is_regex:
                    if re.search(pat, c):
                        new_name = to
                        break
                else:
                    if c == pat:
                        new_name = to
                        break
            except re.error as e:
                logging.warning(f"Invalid regex in column_mappings: {pat} ({e})")
        out.append(new_name)
    return out


def normalize_and_map_df_columns(df: pd.DataFrame, cfg: Dict[str, Any]) -> pd.DataFrame:
    norm_rules = (cfg.get("normalize_columns") or {})
    mapped_rules = (cfg.get("column_mappings") or [])

    # normalize first
    df = df.copy()
    df.columns = [normalize_column_name(c, norm_rules) for c in df.columns]

    # then map/rename
    df.columns = apply_column_mappings(list(df.columns), mapped_rules)
    return df


def enforce_final_column_order(df: pd.DataFrame, cfg: Dict[str, Any]) -> pd.DataFrame:
    order = cfg.get("final_column_order")
    if not order:
        return df
    # Keep only available columns in specified order, then append any extras at end
    ordered_existing = [c for c in order if c in df.columns]
    extras = [c for c in df.columns if c not in ordered_existing]
    return df[ordered_existing + extras]

def parse_dates_inplace(df: pd.DataFrame, cfg: Dict[str, Any]) -> pd.DataFrame:
    dp = cfg.get("date_parsing") or {}
    if not dp or not dp.get("enabled", False):
        return df

    columns = dp.get("columns") or []
    per_col = dp.get("per_column_formats") or {}
    defaults = dp.get("default_formats") or []
    dayfirst = bool(dp.get("dayfirst", False))
    coerce = bool(dp.get("coerce_invalid_to_null", True))
    out_cfg = dp.get("output") or {}
    as_string = bool(out_cfg.get("as_string", True))
    out_fmt = out_cfg.get("format", "%Y-%m-%d")

    for col in columns:
        if col not in df.columns:
            continue
        series = df[col]

        # Skip non-string-like, try conversion anyway
        # Strategy: try per-column formats first, then defaults, finally pandas infer
        parsed = None
        formats_try = per_col.get(col, []) + defaults

        for fmt in formats_try:
            try:
                parsed = pd.to_datetime(series, format=fmt, errors=("coerce" if coerce else "raise"), dayfirst=dayfirst)
                # If we have at least one non-NaT (or no errors), accept
                if parsed.notna().any() or coerce:
                    break
            except Exception:
                parsed = None

        if parsed is None:
            # last resort: infer
            parsed = pd.to_datetime(series, errors=("coerce" if coerce else "raise"), dayfirst=dayfirst, infer_datetime_format=True)

        if as_string:
            # format as string (NaT -> None)
            df[col] = parsed.dt.strftime(out_fmt)
        else:
            df[col] = parsed

    return df

def unify_columns(dfs: List[pd.DataFrame]) -> List[pd.DataFrame]:
    all_cols = set()
    for df in dfs:
        all_cols.update(map(str, df.columns))
    all_cols = list(all_cols)
    unified = []
    for df in dfs:
        df.columns = list(map(str, df.columns))
        unified.append(df.reindex(columns=all_cols))
    return unified


def merge_files(
    cfg: Dict[str, Any],
    input_dir: str,
    include_patterns: List[str],
    sheet_name: Optional[str],
    header_row: Optional[int],
    add_source_columns: bool,
    deduplicate: bool,
    dedup_key_columns: List[str],
    csv_delimiter: str,
    encoding_pref: str,
) -> (pd.DataFrame, pd.DataFrame):
    records = []
    summary_rows = []

    for pattern in include_patterns:
        for path in glob.glob(os.path.join(input_dir, pattern)):
            if os.path.isdir(path):
                continue
            ext = os.path.splitext(path)[1].lower()

            if ext in [".csv", ".tsv"]:
                delim = csv_delimiter if ext == ".csv" else "\t"
                try:
                    df = read_csv_safely(path, delimiter=delim, encoding_pref=encoding_pref)
                    if add_source_columns:
                        df["source_file"] = os.path.basename(path)
                        df["source_sheet"] = ""
                    # normalize/map BEFORE any union to avoid duplicates
                    df = normalize_and_map_df_columns(df, cfg)
                    # parse dates after mapping
                    df = parse_dates_inplace(df, cfg)

                    records.append(df)
                    summary_rows.append({"source_file": os.path.basename(path), "source_sheet": "", "rows": len(df)})
                    logging.info(f"Loaded CSV: {os.path.basename(path)} rows={len(df)}")
                except Exception as e:
                    logging.error(f"Error reading CSV '{path}': {e}\n{traceback.format_exc()}")

            elif ext in [".xlsx", ".xls"]:
                sheets = read_excel_sheets(path, sheet_name, header_row)
                for sname, df in sheets.items():
                    if not isinstance(df, pd.DataFrame):
                        continue
                    if add_source_columns:
                        df["source_file"] = os.path.basename(path)
                        df["source_sheet"] = sname
                    df = normalize_and_map_df_columns(df, cfg)
                    df = parse_dates_inplace(df, cfg)

                    records.append(df)
                    summary_rows.append({"source_file": os.path.basename(path), "source_sheet": sname, "rows": len(df)})
                logging.info(f"Loaded Excel: {os.path.basename(path)} sheets={len(sheets)}")
            else:
                logging.info(f"Skipped unsupported file: {os.path.basename(path)}")

    if not records:
        logging.warning("No data files found to merge.")
        return pd.DataFrame(), pd.DataFrame()

    unified = unify_columns(records)
    merged = pd.concat(unified, ignore_index=True)

    # Final column order (optional)
    merged = enforce_final_column_order(merged, cfg)

    # Deduplication uses POST-MAP column names
    if deduplicate:
        if dedup_key_columns:
            before = len(merged)
            merged = merged.drop_duplicates(subset=dedup_key_columns)
            logging.info(f"Deduplicated by keys {dedup_key_columns}: {before} -> {len(merged)}")
        else:
            before = len(merged)
            merged = merged.drop_duplicates()
            logging.info(f"Deduplicated full rows: {before} -> {len(merged)}")

    summary = pd.DataFrame(summary_rows)
    if not summary.empty:
        summary = summary.groupby(["source_file", "source_sheet"], as_index=False)["rows"].sum()
        summary["total_rows_after_merge"] = len(merged)

    return merged, summary


def save_excel(merged: pd.DataFrame, summary: pd.DataFrame, out_path: str):
    with pd.ExcelWriter(out_path, engine="openpyxl") as writer:
        merged.to_excel(writer, index=False, sheet_name="Merged")
        if not summary.empty:
            summary.to_excel(writer, index=False, sheet_name="Summary")


def main():
    cfg_path = "config.yml"
    cfg = load_config(cfg_path)

    input_dir = cfg.get("input_dir", "./input")
    include_patterns = cfg.get("include_patterns", ["*.xlsx", "*.xls", "*.csv"])
```

```python
def _currency_to_number_series(series: pd.Series) -> pd.Series:
    # Handle: ₹1,23,456.78  |  £1,234.56  |  (123.45) -> -123.45  |  1 234,56
    s = series.astype(str)

    # Detect parentheses (negative)
    neg_mask = s.str.contains(r"^\s*\(.*\)\s*$")
    s = s.str.replace(r"[\(\)]", "", regex=True)

    # Remove currency symbols and common noise
    s = s.str.replace(r"[₹£€$]", "", regex=True)
    s = s.str.replace(r"[^\d,\.\-\s]", "", regex=True)  # keep digits, comma, dot, minus, space
    s = s.str.strip()

    # Remove spaces used as thousand separators
    s = s.str.replace(r"\s+", "", regex=True)

    # Heuristic for comma/dot:
    # If both present, assume comma thousands + dot decimal -> remove commas
    # If only commas present and more than one comma -> treat as thousands -> remove commas
    # Else replace comma with dot if no dot exists (comma-decimal locales)
    both_mask = s.str.contains(",") & s.str.contains(r"\.")
    s = s.where(~both_mask, s.str.replace(",", "", regex=False))

    only_comma_mask = s.str.contains(",") & ~s.str.contains(r"\.")
    multi_comma_mask = only_comma_mask & (s.str.count(",") > 1)
    s = s.where(~multi_comma_mask, s.str.replace(",", "", regex=False))
    # If single comma and no dot: treat comma as decimal
    single_comma_decimal_mask = only_comma_mask & (s.str.count(",") == 1)
    s = s.where(~single_comma_decimal_mask, s.str.replace(",", ".", regex=False))

    # Finally convert
    out = pd.to_numeric(s, errors="coerce")

    # Apply negatives for original parentheses
    out = out.where(~neg_mask, -out)
    return out


def _to_number_series(series: pd.Series) -> pd.Series:
    s = series.astype(str)
    s = s.str.replace(r"[^\d\.\-]", "", regex=True)
    return pd.to_numeric(s, errors="coerce")


def apply_value_transforms(df: pd.DataFrame, cfg: Dict[str, Any]) -> pd.DataFrame:
    vt = cfg.get("value_transforms") or {}
    if not vt.get("enabled", False):
        return df

    ops = vt.get("operations") or []
    if not ops:
        return df

    df = df.copy()

    for op in ops:
        cols = [c for c in _as_iter(op.get("columns")) if c in df.columns]
        actions = op.get("actions") or []
        if not cols or not actions:
            continue

        for col in cols:
            series = df[col]
            for action in actions:
                atype = (action.get("type") or "").lower()

                if atype in ("trim", "ltrim", "rtrim", "uppercase", "lowercase", "titlecase", "strip_non_ascii", "regex_replace"):
                    s = _to_string_series(series)  # operate on strings

                    if atype == "trim":
                        series = s.str.strip()
                    elif atype == "ltrim":
                        series = s.str.lstrip()
                    elif atype == "rtrim":
                        series = s.str.rstrip()
                    elif atype == "uppercase":
                        series = s.str.upper()
                    elif atype == "lowercase":
                        series = s.str.lower()
                    elif atype == "titlecase":
                        series = s.str.title()
                    elif atype == "strip_non_ascii":
                        series = s.map(_strip_non_ascii)
                    elif atype == "regex_replace":
                        pat = action.get("pattern", "")
                        repl = action.get("repl", "")
                        try:
                            series = s.str.replace(pat, repl, regex=True)
                        except re.error as e:
                            logging.warning(f"Bad regex for column '{col}': {pat} ({e})")

                elif atype == "map_values":
                    mapping = action.get("mapping") or {}
                    series = series.map(mapping).where(series.map(mapping).notna(), series)

                elif atype == "fillna":
                    value = action.get("value", "")
                    series = series.fillna(value)
                    # also treat empty strings as NA, then fill
                    series = series.replace("", value)

                elif atype == "to_number":
                    series = _to_number_series(series)

                elif atype == "currency_to_number":
                    series = _currency_to_number_series(series)

                elif atype == "round":
                    nd = action.get("ndigits", 2)
                    try:
                        series = pd.to_numeric(series, errors="coerce").round(nd)
                    except Exception:
                        pass

                elif atype in ("multiply", "divide", "add", "subtract"):
                    factor = action.get("value", 1)
                    series = pd.to_numeric(series, errors="coerce")
                    if atype == "multiply":
                        series = series * factor
                    elif atype == "divide":
                        try:
                            series = series / factor
                        except Exception:
                            series = series  # leave as-is on div by zero etc.
                    elif atype == "add":
                        series = series + factor
                    elif atype == "subtract":
                        series = series - factor

                else:
                    logging.warning(f"Unknown transform '{atype}' on column '{col}' — skipped")

            df[col] = series

    return df

def unify_columns(dfs: List[pd.DataFrame]) -> List[pd.DataFrame]:
    all_cols = set()
    for df in dfs:
        all_cols.update(map(str, df.columns))
    all_cols = list(all_cols)
    unified = []
    for df in dfs:
        df.columns = list(map(str, df.columns))
        unified.append(df.reindex(columns=all_cols))
    return unified


def merge_files(
    cfg: Dict[str, Any],
    input_dir: str,
    include_patterns: List[str],
    sheet_name: Optional[str],
    header_row: Optional[int],
    add_source_columns: bool,
    deduplicate: bool,
    dedup_key_columns: List[str],
    csv_delimiter: str,
    encoding_pref: str,
) -> (pd.DataFrame, pd.DataFrame):
    records = []
    summary_rows = []

    for pattern in include_patterns:
        for path in glob.glob(os.path.join(input_dir, pattern)):
            if os.path.isdir(path):
                continue
            ext = os.path.splitext(path)[1].lower()

            if ext in [".csv", ".tsv"]:
                delim = csv_delimiter if ext == ".csv" else "\t"
                try:
                    df = read_csv_safely(path, delimiter=delim, encoding_pref=encoding_pref)
                    if add_source_columns:
                        df["source_file"] = os.path.basename(path)
                        df["source_sheet"] = ""
                    df = normalize_and_map_df_columns(df, cfg)
                    df = parse_dates_inplace(df, cfg)
                    df = apply_value_transforms(df, cfg)

                    records.append(df)
                    summary_rows.append({"source_file": os.path.basename(path), "source_sheet": "", "rows": len(df)})
                    logging.info(f"Loaded CSV: {os.path.basename(path)} rows={len(df)}")
                except Exception as e:
                    logging.error(f"Error reading CSV '{path}': {e}\n{traceback.format_exc()}")

            elif ext in [".xlsx", ".xls"]:
                sheets = read_excel_sheets(path, sheet_name, header_row)
                for sname, df in sheets.items():
                    if not isinstance(df, pd.DataFrame):
                        continue
                    if add_source_columns:
                        df["source_file"] = os.path.basename(path)
                        df["source_sheet"] = sname
                    df = normalize_and_map_df_columns(df, cfg)
                    df = parse_dates_inplace(df, cfg)
                    df = apply_value_transforms(df, cfg)

                    records.append(df)
                    summary_rows.append({"source_file": os.path.basename(path), "source_sheet": sname, "rows": len(df)})
                logging.info(f"Loaded Excel: {os.path.basename(path)} sheets={len(sheets)}")
            else:
                logging.info(f"Skipped unsupported file: {os.path.basename(path)}")

    if not records:
        logging.warning("No data files found to merge.")
        return pd.DataFrame(), pd.DataFrame()

    unified = unify_columns(records)
    merged = pd.concat(unified, ignore_index=True)

    merged = enforce_final_column_order(merged, cfg)

    if deduplicate:
        if dedup_key_columns:
            before = len(merged)
            merged = merged.drop_duplicates(subset=dedup_key_columns)
            logging.info(f"Deduplicated by keys {dedup_key_columns}: {before} -> {len(merged)}")
        else:
            before = len(merged)
            merged = merged.drop_duplicates()
            logging.info(f"Deduplicated full rows: {before} -> {len(merged)}")

    summary = pd.DataFrame(summary_rows)
    if not summary.empty:
        summary = summary.groupby(["source_file", "source_sheet"], as_index=False)["rows"].sum()
        summary["total_rows_after_merge"] = len(merged)

    return merged, summary


def save_excel(merged: pd.DataFrame, summary: pd.DataFrame, out_path: str):
    with pd.ExcelWriter(out_path, engine="openpyxl") as writer:
        merged.to_excel(writer, index=False, sheet_name="Merged")
        if not summary.empty:
            summary.to_excel(writer, index=False, sheet_name="Summary")


def main():
    cfg_path = "config.yml"
    cfg = load_config(cfg_path)

    input_dir = cfg.get("input_dir", "./input")
```

```python
def detect_encoding(file_path: str, default: str = "utf-8") -> str:

def read_csv_safely(file_path: str, delimiter: str, encoding_pref: str) -> pd.DataFrame:
    enc = detect_encoding(file_path) if encoding_pref == "auto" else encoding_pref
    try:
        return pd.read_csv(file_path, encoding=enc, sep=delimiter)
    except Exception as e:

        return pd.read_csv(file_path, encoding="utf-8", sep=delimiter, engine="python")


def read_excel_sheets(file_path: str, sheet_name: Optional[str], header_row: Optional[int]) -> Dict[str, pd.DataFrame]:
    try:
        if sheet_name and sheet_name != "*":
            df = pd.read_excel(file_path, sheet_name=sheet_name, header=header_row, engine="openpyxl")
            return {sheet_name: df}
        else:

    except Exception as e:

def normalize_column_name(name: Any, rules: Dict[str, Any]) -> str:
    s = str(name)
    if rules.get("strip_spaces", True):
        s = s.strip()
    if rules.get("collapse_inner_spaces", True):
        s = re.sub(r"\s+", " ", s)
    if rules.get("lowercase", False):
        s = s.lower()
    return s


def apply_column_mappings(cols: List[str], mappings: List[Dict[str, Any]]) -> List[str]:
    out = []
    for c in cols:
        new_name = c
        for rule in mappings or []:
            pat = rule.get("match")
            to = rule.get("to", c)
            is_regex = bool(rule.get("regex", False))
            try:
                if is_regex:
                    if re.search(pat, c):
                        new_name = to
                        break
                else:
                    if c == pat:
                        new_name = to
                        break
            except re.error as e:
                logging.warning(f"Invalid regex in column_mappings: {pat} ({e})")
        out.append(new_name)
    return out


def normalize_and_map_df_columns(df: pd.DataFrame, cfg: Dict[str, Any]) -> pd.DataFrame:

    mapped_rules = (cfg.get("column_mappings") or {})

def parse_dates_inplace(df: pd.DataFrame, cfg: Dict[str, Any]) -> pd.DataFrame:
    dp = cfg.get("date_parsing") or {}
    if not dp or not dp.get("enabled", False):
        return df

    columns = dp.get("columns") or []
    per_col = dp.get("per_column_formats") or {}
    defaults = dp.get("default_formats") or []
    dayfirst = bool(dp.get("dayfirst", False))
    coerce = bool(dp.get("coerce_invalid_to_null", True))
    out_cfg = dp.get("output") or {}
    as_string = bool(out_cfg.get("as_string", True))
    out_fmt = out_cfg.get("format", "%Y-%m-%d")

    for col in columns:
        if col not in df.columns:
            continue
        series = df[col]
        parsed = None
        formats_try = per_col.get(col, []) + defaults
        for fmt in formats_try:
            try:
                parsed = pd.to_datetime(series, format=fmt, errors=("coerce" if coerce else "raise"), dayfirst=dayfirst)
                if parsed.notna().any() or coerce:
                    break
            except Exception:
                parsed = None
        if parsed is None:
            parsed = pd.to_datetime(series, errors=("coerce" if coerce else "raise"), dayfirst=dayfirst, infer_datetime_format=True)
        if as_string:
            df[col] = parsed.dt.strftime(out_fmt)
        else:
            df[col] = parsed
    return df

def _currency_to_number_series(series: pd.Series) -> pd.Series:
    s = series.astype(str)
    neg_mask = s.str.contains(r"^\s*\(.*\)\s*$")
    s = s.str.replace(r"[\(\)]", "", regex=True)
    s = s.str.replace(r"[₹£€$]", "", regex=True)
    s = s.str.replace(r"[^\d,\.\-\s]", "", regex=True)
    s = s.str.strip()
    s = s.str.replace(r"\s+", "", regex=True)
    both_mask = s.str.contains(",") & s.str.contains(r"\.")
    s = s.where(~both_mask, s.str.replace(",", "", regex=False))
    only_comma_mask = s.str.contains(",") & ~s.str.contains(r"\.")
    multi_comma_mask = only_comma_mask & (s.str.count(",") > 1)
    s = s.where(~multi_comma_mask, s.str.replace(",", "", regex=False))
    single_comma_decimal_mask = only_comma_mask & (s.str.count(",") == 1)
    s = s.where(~single_comma_decimal_mask, s.str.replace(",", ".", regex=False))
    out = pd.to_numeric(s, errors="coerce")
    out = out.where(~neg_mask, -out)
    return out


def _to_number_series(series: pd.Series) -> pd.Series:
    s = series.astype(str)
    s = s.str.replace(r"[^\d\.\-]", "", regex=True)
    return pd.to_numeric(s, errors="coerce")


def apply_value_transforms(df: pd.DataFrame, cfg: Dict[str, Any]) -> pd.DataFrame:
    vt = cfg.get("value_transforms") or {}
    if not vt.get("enabled", False):
        return df

    ops = vt.get("operations") or []
    if not ops:
        return df

    df = df.copy()

    for op in ops:
        cols = [c for c in _as_iter(op.get("columns")) if c in df.columns]
        actions = op.get("actions") or []
        if not cols or not actions:
            continue

        for col in cols:
            series = df[col]
            for action in actions:
                atype = (action.get("type") or "").lower()

                if atype in ("trim", "ltrim", "rtrim", "uppercase", "lowercase", "titlecase", "strip_non_ascii", "regex_replace"):
                    s = _to_string_series(series)
                    if atype == "trim":
                        series = s.str.strip()
                    elif atype == "ltrim":
                        series = s.str.lstrip()
                    elif atype == "rtrim":
                        series = s.str.rstrip()
                    elif atype == "uppercase":
                        series = s.str.upper()
                    elif atype == "lowercase":
                        series = s.str.lower()
                    elif atype == "titlecase":
                        series = s.str.title()
                    elif atype == "strip_non_ascii":
                        series = s.map(_strip_non_ascii)
                    elif atype == "regex_replace":
                        pat = action.get("pattern", "")
                        repl = action.get("repl", "")
                        try:
                            series = s.str.replace(pat, repl, regex=True)
                        except re.error as e:
                            logging.warning(f"Bad regex for column '{col}': {pat} ({e})")

                elif atype == "map_values":
                    mapping = action.get("mapping") or {}
                    mapped = series.map(mapping)
                    series = mapped.where(mapped.notna(), series)

                elif atype == "fillna":
                    value = action.get("value", "")
                    series = series.replace("", pd.NA)
                    series = series.fillna(value)

                elif atype == "to_number":
                    series = _to_number_series(series)

                elif atype == "currency_to_number":
                    series = _currency_to_number_series(series)

                elif atype == "round":
                    nd = action.get("ndigits", 2)
                    series = pd.to_numeric(series, errors="coerce").round(nd)

                elif atype in ("multiply", "divide", "add", "subtract"):
                    factor = action.get("value", 1)
                    series = pd.to_numeric(series, errors="coerce")
                    if atype == "multiply":
                        series = series * factor
                    elif atype == "divide":
                        series = series / factor
                    elif atype == "add":
                        series = series + factor
                    elif atype == "subtract":
                        series = series - factor

                else:
                    logging.warning(f"Unknown transform '{atype}' on column '{col}' — skipped")

            df[col] = series

    return df

def filter_sheets(file_path: str, sheets: Dict[str, pd.DataFrame], cfg: Dict[str, Any]) -> Dict[str, pd.DataFrame]:
    rules = cfg.get("sheet_filters") or {}
    inc = set(rules.get("include", []) or [])
    exc = set(rules.get("exclude", []) or [])
    inc_rx = rules.get("include_regex", []) or []
    exc_rx = rules.get("exclude_regex", []) or []

    if not (inc or exc or inc_rx or exc_rx):
        return sheets  # no filtering

    out: Dict[str, pd.DataFrame] = {}
    for sname, df in sheets.items():
        name = str(sname)

        # Include logic
        include_ok = True
        if inc or inc_rx:
            include_ok = (name in inc) or _match_any_regex(name, inc_rx)

        # Exclude logic (only if included)
        if include_ok and (name in exc or _match_any_regex(name, exc_rx)):
            include_ok = False

        if include_ok:
            out[name] = df

    logging.info(f"Sheet filter on '{os.path.basename(file_path)}': kept {list(out.keys()) or 'none'}")
    return out

def apply_computed_columns(df: pd.DataFrame, cfg: Dict[str, Any]) -> pd.DataFrame:
    cc = cfg.get("computed_columns") or {}
    exprs = cc.get("expressions") or []
    if not exprs:

    df = df.copy()

            # Support "col = expr" or pure expr with assignment(s)
            if "=" in expr:

            else:

                df.eval(expr, engine="python", inplace=True)
            logging.info(f"Computed: {expr}")
        except Exception as e:
            logging.warning(f"Failed computed expression '{expr}': {e}")
    return df


def _apply_rule(df: pd.DataFrame, rule: Dict[str, Any]) -> pd.Series:
    col = rule.get("column")
    op = (rule.get("op") or "").lower()
    val = rule.get("value", None)

    if col not in df.columns:
        logging.warning(f"Filter rule skipped: column '{col}' not in DataFrame")
        return pd.Series([True] * len(df), index=df.index)

    s = df[col]

    if op == "notnull":
        return s.notna()
    if op == "isnull":
        return s.isna()

    if op in (">", ">=", "<", "<=", "==", "!="):
        try:
            return eval(f"s {op} @val")
        except Exception:
            # Try numeric
            try:
                sn = pd.to_numeric(s, errors="coerce")
                return eval(f"sn {op} @val")
            except Exception as e:
                logging.warning(f"Rule '{col} {op} {val}' failed: {e}")
                return pd.Series([True] * len(df), index=df.index)

    if op == "contains":
        return s.astype(str).str.contains(str(val), na=False, regex=True)
    if op == "in":
        return s.isin(val if isinstance(val, list) else [val])
    if op == "notin":
        return ~s.isin(val if isinstance(val, list) else [val])

    logging.warning(f"Unknown filter op '{op}' in rule on '{col}'")
    return pd.Series([True] * len(df), index=df.index)


def apply_global_row_filters(df: pd.DataFrame, cfg: Dict[str, Any]) -> pd.DataFrame:
    gf = cfg.get("global_row_filters") or {}
    queries: List[str] = gf.get("queries") or []
    rules: List[Dict[str, Any]] = gf.get("rules") or []
    dropna = gf.get("drop_na") or {}

    out = df.copy()

    # Apply query strings in order
    for q in queries:
        try:
            before = len(out)
            out = out.query(q, engine="python")
            logging.info(f"Filter query '{q}': {before} -> {len(out)}")
        except Exception as e:
            logging.warning(f"Query '{q}' failed: {e}")

    # Apply rules (AND across all rules)
    if rules:
        mask = pd.Series([True] * len(out), index=out.index)
        for r in rules:
            try:
                mask &= _apply_rule(out, r)
            except Exception as e:
                logging.warning(f"Rule failed {r}: {e}")
        before = len(out)
        out = out[mask]
        logging.info(f"Filter rules applied: {before} -> {len(out)}")

    # Optional NA drop
    cols = dropna.get("columns") or []
    if cols:
        how = dropna.get("how", "any")
        before = len(out)
        out = out.dropna(subset=[c for c in cols if c in out.columns], how=how)
        logging.info(f"drop_na({cols}, how={how}): {before} -> {len(out)}")

    return out

def unify_columns(dfs: List[pd.DataFrame]) -> List[pd.DataFrame]:
    all_cols = set()
    for df in dfs:
        all_cols.update(map(str, df.columns))
    all_cols = list(all_cols)
    unified = []
    for df in dfs:
        df.columns = list(map(str, df.columns))
        unified.append(df.reindex(columns=all_cols))
    return unified


def merge_files(
    cfg: Dict[str, Any],
    input_dir: str,
    include_patterns: List[str],
    sheet_name: Optional[str],
    header_row: Optional[int],
    add_source_columns: bool,
    deduplicate: bool,
    dedup_key_columns: List[str],
    csv_delimiter: str,
    encoding_pref: str,
) -> (pd.DataFrame, pd.DataFrame):
    records = []
    summary_rows = []

    for pattern in include_patterns:
        for path in glob.glob(os.path.join(input_dir, pattern)):
            if os.path.isdir(path):
                continue
            ext = os.path.splitext(path)[1].lower()

            if ext in [".csv", ".tsv"]:
                delim = csv_delimiter if ext == ".csv" else "\t"
                try:
                    df = read_csv_safely(path, delimiter=delim, encoding_pref=encoding_pref)
                    if add_source_columns:
                        df["source_file"] = os.path.basename(path)
                        df["source_sheet"] = ""
                    df = normalize_and_map_df_columns(df, cfg)
                    df = parse_dates_inplace(df, cfg)
                    df = apply_value_transforms(df, cfg)

                    records.append(df)
                    summary_rows.append({"source_file": os.path.basename(path), "source_sheet": "", "rows": len(df)})
                    logging.info(f"Loaded CSV: {os.path.basename(path)} rows={len(df)}")
                except Exception as e:
                    logging.error(f"Error reading CSV '{path}': {e}\n{traceback.format_exc()}")

            elif ext in [".xlsx", ".xls"]:
                sheets = read_excel_sheets(path, sheet_name, header_row)
                sheets = filter_sheets(path, sheets, cfg)  # NEW
                for sname, df in sheets.items():
                    if not isinstance(df, pd.DataFrame):
                        continue
                    if add_source_columns:
                        df["source_file"] = os.path.basename(path)
                        df["source_sheet"] = sname
                    df = normalize_and_map_df_columns(df, cfg)
                    df = parse_dates_inplace(df, cfg)
                    df = apply_value_transforms(df, cfg)

                    records.append(df)
                    summary_rows.append({"source_file": os.path.basename(path), "source_sheet": sname, "rows": len(df)})
                logging.info(f"Loaded Excel: {os.path.basename(path)} sheets={len(sheets)}")
            else:
                logging.info(f"Skipped unsupported file: {os.path.basename(path)}")

    if not records:
        logging.warning("No data files found to merge.")
        return pd.DataFrame(), pd.DataFrame()

    unified = unify_columns(records)
    merged = pd.concat(unified, ignore_index=True)

    # Optional final order before computed columns/filters
    merged = enforce_final_column_order(merged, cfg)

    # Computed columns
    cc_when = (cfg.get("computed_columns") or {}).get("when", "before").lower()
    if cc_when == "before":
        merged = apply_computed_columns(merged, cfg)

    # Deduplicate BEFORE or AFTER filters? Keeping as before: before filters.
    deduplicate = bool(deduplicate)
    if deduplicate:
        if dedup_key_columns:
            before = len(merged)
            merged = merged.drop_duplicates(subset=dedup_key_columns)
            logging.info(f"Deduplicated by keys {dedup_key_columns}: {before} -> {len(merged)}")
        else:
            before = len(merged)
            merged = merged.drop_duplicates()
            logging.info(f"Deduplicated full rows: {before} -> {len(merged)}")

    # Global row filters
    merged = apply_global_row_filters(merged, cfg)

    # Computed columns AFTER filters if configured so
    if cc_when == "after":
        merged = apply_computed_columns(merged, cfg)

    # Update summary
    summary = pd.DataFrame(summary_rows)
    if not summary.empty:
        summary = summary.groupby(["source_file", "source_sheet"], as_index=False)["rows"].sum()
        summary["total_rows_after_merge"] = len(merged)

    return merged, summary


def save_excel(merged: pd.DataFrame, summary: pd.DataFrame, out_path: str):
    with pd.ExcelWriter(out_path, engine="openpyxl") as writer:
        merged.to_excel(writer, index=False, sheet_name="Merged")
        if not summary.empty:
            summary.to_excel(writer, index=False, sheet_name="Summary")
```

```python
def detect_encoding(file_path: str, default: str = "utf-8") -> str:
    if not chardet:
        return default
    try:
        with open(file_path, "rb") as f:
            raw = f.read(50000)
            res = chardet.detect(raw)
            return res.get("encoding") or default
    except Exception:
        return default


def read_csv_safely(file_path: str, delimiter: str, encoding_pref: str) -> pd.DataFrame:
    enc = detect_encoding(file_path) if encoding_pref == "auto" else encoding_pref
    try:
        return pd.read_csv(file_path, encoding=enc, sep=delimiter)
    except Exception as e:
        logging.warning(f"CSV read failed with {enc}: {e}. Retrying with utf-8.")
        return pd.read_csv(file_path, encoding="utf-8", sep=delimiter, engine="python")


def read_excel_sheets(file_path: str, sheet_name: Optional[str], header_row: Optional[int]) -> Dict[str, pd.DataFrame]:
    try:
        if sheet_name and sheet_name != "*":
            df = pd.read_excel(file_path, sheet_name=sheet_name, header=header_row, engine="openpyxl")
            return {sheet_name: df}
        else:
            dfs = pd.read_excel(file_path, sheet_name=None, header=header_row, engine="openpyxl")
            return dfs
    except Exception as e:
        logging.error(f"Failed reading Excel '{file_path}': {e}")
        return {}

def normalize_column_name(name: Any, rules: Dict[str, Any]) -> str:
    s = str(name)
    if rules.get("strip_spaces", True):
        s = s.strip()
    if rules.get("collapse_inner_spaces", True):
        s = re.sub(r"\s+", " ", s)
    if rules.get("lowercase", False):
        s = s.lower()
    return s


def apply_column_mappings(cols: List[str], mappings: List[Dict[str, Any]]) -> List[str]:
    out = []
    for c in cols:
        new_name = c
        for rule in mappings or []:
            pat = rule.get("match")
            to = rule.get("to", c)
            is_regex = bool(rule.get("regex", False))
            try:
                if is_regex:
                    if re.search(pat, c):
                        new_name = to
                        break
                else:
                    if c == pat:
                        new_name = to
                        break
            except re.error as e:
                logging.warning(f"Invalid regex in column_mappings: {pat} ({e})")
        out.append(new_name)
    return out


def normalize_and_map_df_columns(df: pd.DataFrame, cfg: Dict[str, Any]) -> pd.DataFrame:
    norm_rules = (cfg.get("normalize_columns") or {})
    mapped_rules = (cfg.get("column_mappings") or [])

    df = df.copy()
    df.columns = [normalize_column_name(c, norm_rules) for c in df.columns]
    df.columns = apply_column_mappings(list(df.columns), mapped_rules)
    return df


def enforce_final_column_order(df: pd.DataFrame, cfg: Dict[str, Any]) -> pd.DataFrame:
    order = cfg.get("final_column_order")
    if not order:
        return df
    ordered_existing = [c for c in order if c in df.columns]
    extras = [c for c in df.columns if c not in ordered_existing]
    return df[ordered_existing + extras]

def parse_dates_inplace(df: pd.DataFrame, cfg: Dict[str, Any]) -> pd.DataFrame:
    dp = cfg.get("date_parsing") or {}
    if not dp or not dp.get("enabled", False):
        return df

    columns = dp.get("columns") or []
    per_col = dp.get("per_column_formats") or {}
    defaults = dp.get("default_formats") or []
    dayfirst = bool(dp.get("dayfirst", False))
    coerce = bool(dp.get("coerce_invalid_to_null", True))
    out_cfg = dp.get("output") or {}
    as_string = bool(out_cfg.get("as_string", True))
    out_fmt = out_cfg.get("format", "%Y-%m-%d")

    for col in columns:
        if col not in df.columns:
            continue
        series = df[col]
        parsed = None
        formats_try = per_col.get(col, []) + defaults
        for fmt in formats_try:
            try:
                parsed = pd.to_datetime(series, format=fmt, errors=("coerce" if coerce else "raise"), dayfirst=dayfirst)
                if parsed.notna().any() or coerce:
                    break
            except Exception:
                parsed = None
        if parsed is None:
            parsed = pd.to_datetime(series, errors=("coerce" if coerce else "raise"), dayfirst=dayfirst, infer_datetime_format=True)
        if as_string:
            df[col] = parsed.dt.strftime(out_fmt)
        else:
            df[col] = parsed
    return df

def _as_iter(obj: Any) -> Iterable:
    if obj is None:
        return []
    if isinstance(obj, (list, tuple, set)):
        return obj
    return [obj]


def _strip_non_ascii(s: str) -> str:
    try:
        return s.encode("ascii", "ignore").decode("ascii")
    except Exception:
        return s


def _to_string_series(series: pd.Series) -> pd.Series:
    return series.astype(str)


def _currency_to_number_series(series: pd.Series) -> pd.Series:
    s = series.astype(str)
    neg_mask = s.str.contains(r"^\s*\(.*\)\s*$")
    s = s.str.replace(r"[\(\)]", "", regex=True)
    s = s.str.replace(r"[₹£€$]", "", regex=True)
    s = s.str.replace(r"[^\d,\.\-\s]", "", regex=True)
    s = s.str.strip()
    s = s.str.replace(r"\s+", "", regex=True)
    both_mask = s.str.contains(",") & s.str.contains(r"\.")
    s = s.where(~both_mask, s.str.replace(",", "", regex=False))
    only_comma_mask = s.str.contains(",") & ~s.str.contains(r"\.")
    multi_comma_mask = only_comma_mask & (s.str.count(",") > 1)
    s = s.where(~multi_comma_mask, s.str.replace(",", "", regex=False))
    single_comma_decimal_mask = only_comma_mask & (s.str.count(",") == 1)
    s = s.where(~single_comma_decimal_mask, s.str.replace(",", ".", regex=False))
    out = pd.to_numeric(s, errors="coerce")
    out = out.where(~neg_mask, -out)
    return out


def _to_number_series(series: pd.Series) -> pd.Series:
    s = series.astype(str)
    s = s.str.replace(r"[^\d\.\-]", "", regex=True)
    return pd.to_numeric(s, errors="coerce")


def apply_value_transforms(df: pd.DataFrame, cfg: Dict[str, Any]) -> pd.DataFrame:
    vt = cfg.get("value_transforms") or {}
    if not vt.get("enabled", False):
        return df

    ops = vt.get("operations") or []
    if not ops:
        return df

    df = df.copy()

    for op in ops:
        cols = [c for c in _as_iter(op.get("columns")) if c in df.columns]
        actions = op.get("actions") or []
        if not cols or not actions:
            continue

        for col in cols:
            series = df[col]
            for action in actions:
                atype = (action.get("type") or "").lower()

                if atype in ("trim", "ltrim", "rtrim", "uppercase", "lowercase", "titlecase", "strip_non_ascii", "regex_replace"):
                    s = _to_string_series(series)
                    if atype == "trim":
                        series = s.str.strip()
                    elif atype == "ltrim":
                        series = s.str.lstrip()
                    elif atype == "rtrim":
                        series = s.str.rstrip()
                    elif atype == "uppercase":
                        series = s.str.upper()
                    elif atype == "lowercase":
                        series = s.str.lower()
                    elif atype == "titlecase":
                        series = s.str.title()
                    elif atype == "strip_non_ascii":
                        series = s.map(_strip_non_ascii)
                    elif atype == "regex_replace":
                        pat = action.get("pattern", "")
                        repl = action.get("repl", "")
                        try:
                            series = s.str.replace(pat, repl, regex=True)
                        except re.error as e:
                            logging.warning(f"Bad regex for column '{col}': {pat} ({e})")

                elif atype == "map_values":
                    mapping = action.get("mapping") or {}
                    mapped = series.map(mapping)
                    series = mapped.where(mapped.notna(), series)

                elif atype == "fillna":
                    value = action.get("value", "")
                    series = series.replace("", pd.NA)
                    series = series.fillna(value)

                elif atype == "to_number":
                    series = _to_number_series(series)

                elif atype == "currency_to_number":
                    series = _currency_to_number_series(series)

                elif atype == "round":
                    nd = action.get("ndigits", 2)
                    series = pd.to_numeric(series, errors="coerce").round(nd)

                elif atype in ("multiply", "divide", "add", "subtract"):
                    factor = action.get("value", 1)
                    series = pd.to_numeric(series, errors="coerce")
                    if atype == "multiply":
                        series = series * factor
                    elif atype == "divide":
                        series = series / factor
                    elif atype == "add":
                        series = series + factor
                    elif atype == "subtract":
                        series = series - factor

                else:
                    logging.warning(f"Unknown transform '{atype}' on column '{col}' — skipped")

            df[col] = series

    return df

def _match_any_regex(name: str, patterns: List[str]) -> bool:
    for pat in patterns or []:
        try:
            if re.search(pat, name):
                return True
        except re.error as e:
            logging.warning(f"Invalid sheet regex '{pat}': {e}")
    return False


def filter_sheets(file_path: str, sheets: Dict[str, pd.DataFrame], cfg: Dict[str, Any]) -> Dict[str, pd.DataFrame]:
    rules = cfg.get("sheet_filters") or {}
    inc = set(rules.get("include", []) or [])
    exc = set(rules.get("exclude", []) or [])
    inc_rx = rules.get("include_regex", []) or []
    exc_rx = rules.get("exclude_regex", []) or []

    if not (inc or exc or inc_rx or exc_rx):
        return sheets

    out: Dict[str, pd.DataFrame] = {}
    for sname, df in sheets.items():
        name = str(sname)
        include_ok = True
        if inc or inc_rx:
            include_ok = (name in inc) or _match_any_regex(name, inc_rx)
        if include_ok and (name in exc or _match_any_regex(name, exc_rx)):
            include_ok = False
        if include_ok:
            out[name] = df

    logging.info(f"Sheet filter on '{os.path.basename(file_path)}': kept {list(out.keys()) or 'none'}")
    return out

def apply_type_hints(df: pd.DataFrame, cfg: Dict[str, Any]) -> pd.DataFrame:
    th = cfg.get("type_hints") or {}
    if not th.get("enabled", False):
        return df
    cols_map: Dict[str, str] = th.get("columns") or {}
    errors = th.get("errors", "coerce")
    df = df.copy()

    for col, t in cols_map.items():
        if col not in df.columns:
            continue
        t = (str(t) or "").lower()
        try:
            if t in ("number", "float", "double"):
                df[col] = pd.to_numeric(df[col], errors=errors)
            elif t in ("integer", "int"):
                df[col] = pd.to_numeric(df[col], errors=errors).astype("Int64")
            elif t in ("string", "text"):
                df[col] = df[col].astype("string")
            elif t in ("boolean", "bool"):
                # handle common truthy/falsey strings
                s = df[col].astype(str).str.strip().str.lower()
                map_bool = {"true": True, "t": True, "1": True, "yes": True, "y": True,
                            "false": False, "f": False, "0": False, "no": False, "n": False, "": pd.NA, "nan": pd.NA}
                df[col] = s.map(map_bool).astype("boolean")
            elif t in ("category", "categorical"):
                df[col] = df[col].astype("category")
            else:
                logging.warning(f"type_hints: unsupported type '{t}' for column '{col}', skipped")
        except Exception as e:
            logging.warning(f"type_hints failed on '{col}' as '{t}': {e}")

    return df

def _ensure_date_bucket(df: pd.DataFrame, item: Dict[str, Any], cfg: Dict[str, Any]) -> pd.DataFrame:
    db = item.get("date_bucket")
    if not db:
        return df
    col = db.get("column")
    if not col or col not in df.columns:
        return df
    freq = db.get("freq", "MS")
    label = db.get("label", "bucket")
    parse_if_needed = bool(db.get("parse_if_needed", True))

    series = df[col]
    if not pd.api.types.is_datetime64_any_dtype(series):
        if parse_if_needed:
            # respect dayfirst from date_parsing if present
            dp = cfg.get("date_parsing") or {}
            dayfirst = bool(dp.get("dayfirst", False))
            series = pd.to_datetime(series, errors="coerce", dayfirst=dayfirst, infer_datetime_format=True)
        else:
            return df

    df = df.copy()
    df[label] = series.dt.to_period(freq).dt.to_timestamp()  # use period start as timestamp
    return df


def build_grouping_summaries(df: pd.DataFrame, cfg: Dict[str, Any]) -> Dict[str, pd.DataFrame]:
    gs = cfg.get("grouping_summaries") or {}
    if not gs.get("enabled", False):
        return {}
    out: Dict[str, pd.DataFrame] = {}
    items = gs.get("items") or []

    for item in items:
        name = item.get("name") or "Summary"
        work = _ensure_date_bucket(df, item, cfg)
        group_by = item.get("group_by") or []
        aggs = item.get("aggregations") or {}
        sort_by = item.get("sort_by") or []

        if not group_by or not aggs:
            logging.warning(f"grouping_summaries item '{name}': requires group_by and aggregations")
            continue

        # Build agg dict
        agg_dict: Dict[str, List[str]] = {}
        for col, funcs in aggs.items():
            if isinstance(funcs, str):
                agg_dict[col] = [funcs]
            elif isinstance(funcs, list):
                agg_dict[col] = funcs
            else:
                logging.warning(f"Invalid aggregation spec for '{col}' in '{name}'")
        try:
            summary = work.groupby([g for g in group_by if g in work.columns]).agg(agg_dict)
            # Flatten MultiIndex columns
            summary.columns = ["_".join([c for c in tup if c]) for tup in summary.columns.to_flat_index()]
            summary = summary.reset_index()
            if sort_by:
                sort_cols = [c for c in sort_by if c in summary.columns]
                if sort_cols:
                    summary = summary.sort_values(by=sort_cols)
            out[name] = summary
            logging.info(f"Built grouping summary '{name}' with {len(summary)} rows")
        except Exception as e:
            logging.warning(f"Failed grouping summary '{name}': {e}")

    return out

def unify_columns(dfs: List[pd.DataFrame]) -> List[pd.DataFrame]:
    all_cols = set()
    for df in dfs:
        all_cols.update(map(str, df.columns))
    all_cols = list(all_cols)
    unified = []
    for df in dfs:
        df.columns = list(map(str, df.columns))
        unified.append(df.reindex(columns=all_cols))
    return unified


def merge_files(
    cfg: Dict[str, Any],
    input_dir: str,
    include_patterns: List[str],
    sheet_name: Optional[str],
    header_row: Optional[int],
    add_source_columns: bool,
    deduplicate: bool,
    dedup_key_columns: List[str],
    csv_delimiter: str,
    encoding_pref: str,
) -> (pd.DataFrame, pd.DataFrame, Dict[str, pd.DataFrame]):
    records = []
    summary_rows = []

    for pattern in include_patterns:
        for path in glob.glob(os.path.join(input_dir, pattern)):
            if os.path.isdir(path):
                continue
            ext = os.path.splitext(path)[1].lower()

            if ext in [".csv", ".tsv"]:
                delim = csv_delimiter if ext == ".csv" else "\t"
                try:
                    df = read_csv_safely(path, delimiter=delim, encoding_pref=encoding_pref)
                    if add_source_columns:
                        df["source_file"] = os.path.basename(path)
                        df["source_sheet"] = ""
                    df = normalize_and_map_df_columns(df, cfg)
                    df = parse_dates_inplace(df, cfg)
                    df = apply_value_transforms(df, cfg)
                    df = apply_type_hints(df, cfg)

                    records.append(df)
                    summary_rows.append({"source_file": os.path.basename(path), "source_sheet": "", "rows": len(df)})
                    logging.info(f"Loaded CSV: {os.path.basename(path)} rows={len(df)}")
                except Exception as e:
                    logging.error(f"Error reading CSV '{path}': {e}\n{traceback.format_exc()}")

            elif ext in [".xlsx", ".xls"]:
                sheets = read_excel_sheets(path, sheet_name, header_row)
                sheets = filter_sheets(path, sheets, cfg)
                for sname, df in sheets.items():
                    if not isinstance(df, pd.DataFrame):
                        continue
                    if add_source_columns:
                        df["source_file"] = os.path.basename(path)
                        df["source_sheet"] = sname
                    df = normalize_and_map_df_columns(df, cfg)
                    df = parse_dates_inplace(df, cfg)
                    df = apply_value_transforms(df, cfg)
                    df = apply_type_hints(df, cfg)

                    records.append(df)
                    summary_rows.append({"source_file": os.path.basename(path), "source_sheet": sname, "rows": len(df)})
                logging.info(f"Loaded Excel: {os.path.basename(path)} sheets={len(sheets)}")
            else:
                logging.info(f"Skipped unsupported file: {os.path.basename(path)}")

    if not records:
        logging.warning("No data files found to merge.")
        return pd.DataFrame(), pd.DataFrame(), {}

    unified = unify_columns(records)
    merged = pd.concat(unified, ignore_index=True)

    # Optional final order before computed columns/filters
    merged = enforce_final_column_order(merged, cfg)

    # Computed columns
    cc_when = (cfg.get("computed_columns") or {}).get("when", "before").lower()
    if cc_when == "before":
        merged = apply_computed_columns(merged, cfg)

    # Deduplicate (pre-filters)
    if deduplicate:
        if dedup_key_columns:
            before = len(merged)
            merged = merged.drop_duplicates(subset=dedup_key_columns)
            logging.info(f"Deduplicated by keys {dedup_key_columns}: {before} -> {len(merged)}")
        else:
            before = len(merged)
            merged = merged.drop_duplicates()
            logging.info(f"Deduplicated full rows: {before} -> {len(merged)}")

    # Global row filters
    merged = apply_global_row_filters(merged, cfg)

    # Computed columns AFTER filters if configured
    if cc_when == "after":
        merged = apply_computed_columns(merged, cfg)

    # Build grouping summaries from the final merged df
    group_sheets = build_grouping_summaries(merged, cfg)

    # Update summary
    summary = pd.DataFrame(summary_rows)
    if not summary.empty:
        summary = summary.groupby(["source_file", "source_sheet"], as_index=False)["rows"].sum()
        summary["total_rows_after_merge"] = len(merged)

    return merged, summary, group_sheets

def apply_computed_columns(df: pd.DataFrame, cfg: Dict[str, Any]) -> pd.DataFrame:
    cc = cfg.get("computed_columns") or {}
    exprs = cc.get("expressions") or []
    if not exprs:
        return df
    df = df.copy()
    for expr in exprs:
        try:
            if "=" in expr:
                df = df.eval(expr, engine="python")
            else:
                df.eval(expr, engine="python", inplace=True)
            logging.info(f"Computed: {expr}")
        except Exception as e:
            logging.warning(f"Failed computed expression '{expr}': {e}")
    return df


def _apply_rule(df: pd.DataFrame, rule: Dict[str, Any]) -> pd.Series:
    col = rule.get("column")
    op = (rule.get("op") or "").lower()
    val = rule.get("value", None)

    if col not in df.columns:
        logging.warning(f"Filter rule skipped: column '{col}' not in DataFrame")
        return pd.Series([True] * len(df), index=df.index)

    s = df[col]

    if op == "notnull":
        return s.notna()
    if op == "isnull":
        return s.isna()

    if op in (">", ">=", "<", "<=", "==", "!="):
        try:
            return eval(f"s {op} @val")
        except Exception:
            try:
                sn = pd.to_numeric(s, errors="coerce")
                return eval(f"sn {op} @val")
            except Exception as e:
                logging.warning(f"Rule '{col} {op} {val}' failed: {e}")
                return pd.Series([True] * len(df), index=df.index)

    if op == "contains":
        return s.astype(str).str.contains(str(val), na=False, regex=True)
    if op == "in":
        return s.isin(val if isinstance(val, list) else [val])
    if op == "notin":
        return ~s.isin(val if isinstance(val, list) else [val])

    logging.warning(f"Unknown filter op '{op}' in rule on '{col}'")
    return pd.Series([True] * len(df), index=df.index)


def apply_global_row_filters(df: pd.DataFrame, cfg: Dict[str, Any]) -> pd.DataFrame:
    gf = cfg.get("global_row_filters") or {}
    queries: List[str] = gf.get("queries") or []
    rules: List[Dict[str, Any]] = gf.get("rules") or []
    dropna = gf.get("drop_na") or {}

    out = df.copy()

    for q in queries:
        try:
            before = len(out)
            out = out.query(q, engine="python")
            logging.info(f"Filter query '{q}': {before} -> {len(out)}")
        except Exception as e:
            logging.warning(f"Query '{q}' failed: {e}")

    if rules:
        mask = pd.Series([True] * len(out), index=out.index)
        for r in rules:
            try:
                mask &= _apply_rule(out, r)
            except Exception as e:
                logging.warning(f"Rule failed {r}: {e}")
        before = len(out)
        out = out[mask]
        logging.info(f"Filter rules applied: {before} -> {len(out)}")

    cols = dropna.get("columns") or []
    if cols:
        how = dropna.get("how", "any")
        before = len(out)
        out = out.dropna(subset=[c for c in cols if c in out.columns], how=how)
        logging.info(f"drop_na({cols}, how={how}): {before} -> {len(out)}")

    return out

def save_excel(merged: pd.DataFrame, summary: pd.DataFrame, group_sheets: Dict[str, pd.DataFrame], out_path: str):
    with pd.ExcelWriter(out_path, engine="openpyxl") as writer:
        merged.to_excel(writer, index=False, sheet_name="Merged")
        if not summary.empty:

        # extra grouping sheets
        for name, df in group_sheets.items():
            safe = re.sub(r"[^A-Za-z0-9_]", "_", name)[:31] or "Group"
            df.to_excel(writer, index=False, sheet_name=safe)
```

```python
def normalize_column_name(name: Any, rules: Dict[str, Any]) -> str:
    s = str(name)

        s = s.strip()
    if rules.get("collapse_inner_spaces", True):
        s = re.sub(r"\s+", " ", s)

def apply_column_mappings(cols: List[str], mappings: List[Dict[str, Any]]) -> List[str]:
    out = []
    for c in cols:
        new_name = c
        for rule in mappings or []:
            pat = rule.get("match")
            to = rule.get("to", c)
            is_regex = bool(rule.get("regex", False))
            try:
                if is_regex:
                    if re.search(pat, c):
                        new_name = to
                        break
                else:
                    if c == pat:
                        new_name = to
                        break
            except re.error as e:

        out.append(new_name)
    return out

def parse_dates_inplace(df: pd.DataFrame, cfg: Dict[str, Any]) -> pd.DataFrame:
    dp = cfg.get("date_parsing") or {}
    if not dp or not dp.get("enabled", False):
        return df

    columns = dp.get("columns") or []
    per_col = dp.get("per_column_formats") or {}
    defaults = dp.get("default_formats") or []
    dayfirst = bool(dp.get("dayfirst", False))
    coerce = bool(dp.get("coerce_invalid_to_null", True))
    out_cfg = dp.get("output") or {}
    as_string = bool(out_cfg.get("as_string", True))
    out_fmt = out_cfg.get("format", "%Y-%m-%d")

    for col in columns:
        if col not in df.columns:
            continue
        series = df[col]
        parsed = None
        formats_try = per_col.get(col, []) + defaults
        for fmt in formats_try:
            try:
                parsed = pd.to_datetime(series, format=fmt, errors=("coerce" if coerce else "raise"), dayfirst=dayfirst)
                if parsed.notna().any() or coerce:
                    break
            except Exception:
                parsed = None
        if parsed is None:
            parsed = pd.to_datetime(series, errors=("coerce" if coerce else "raise"), dayfirst=dayfirst, infer_datetime_format=True)
        if as_string:
            df[col] = parsed.dt.strftime(out_fmt)
        else:
            df[col] = parsed
    return df

def _currency_to_number_series(series: pd.Series) -> pd.Series:
    s = series.astype(str)
    neg_mask = s.str.contains(r"^\s*\(.*\)\s*$")
    s = s.str.replace(r"[\(\)]", "", regex=True)
    s = s.str.replace(r"[₹£€$]", "", regex=True)
    s = s.str.replace(r"[^\d,\.\-\s]", "", regex=True)
    s = s.str.strip()
    s = s.str.replace(r"\s+", "", regex=True)
    both_mask = s.str.contains(",") & s.str.contains(r"\.")
    s = s.where(~both_mask, s.str.replace(",", "", regex=False))
    only_comma_mask = s.str.contains(",") & ~s.str.contains(r"\.")
    multi_comma_mask = only_comma_mask & (s.str.count(",") > 1)
    s = s.where(~multi_comma_mask, s.str.replace(",", "", regex=False))
    single_comma_decimal_mask = only_comma_mask & (s.str.count(",") == 1)
    s = s.where(~single_comma_decimal_mask, s.str.replace(",", ".", regex=False))
    out = pd.to_numeric(s, errors="coerce")
    out = out.where(~neg_mask, -out)
    return out


def _to_number_series(series: pd.Series) -> pd.Series:
    s = series.astype(str)
    s = s.str.replace(r"[^\d\.\-]", "", regex=True)
    return pd.to_numeric(s, errors="coerce")


def apply_value_transforms(df: pd.DataFrame, cfg: Dict[str, Any]) -> pd.DataFrame:
    vt = cfg.get("value_transforms") or {}
    if not vt.get("enabled", False):
        return df

    ops = vt.get("operations") or []
    if not ops:
        return df

    df = df.copy()

    for op in ops:
        cols = [c for c in _as_iter(op.get("columns")) if c in df.columns]
        actions = op.get("actions") or []
        if not cols or not actions:
            continue

        for col in cols:
            series = df[col]
            for action in actions:
                atype = (action.get("type") or "").lower()

                if atype in ("trim", "ltrim", "rtrim", "uppercase", "lowercase", "titlecase", "strip_non_ascii", "regex_replace"):
                    s = _to_string_series(series)
                    if atype == "trim":
                        series = s.str.strip()
                    elif atype == "ltrim":
                        series = s.str.lstrip()
                    elif atype == "rtrim":
                        series = s.str.rstrip()
                    elif atype == "uppercase":
                        series = s.str.upper()
                    elif atype == "lowercase":
                        series = s.str.lower()
                    elif atype == "titlecase":
                        series = s.str.title()
                    elif atype == "strip_non_ascii":
                        series = s.map(_strip_non_ascii)
                    elif atype == "regex_replace":
                        pat = action.get("pattern", "")
                        repl = action.get("repl", "")
                        try:
                            series = s.str.replace(pat, repl, regex=True)
                        except re.error as e:
                            logging.warning(f"Bad regex for column '{col}': {pat} ({e})")

                elif atype == "map_values":
                    mapping = action.get("mapping") or {}
                    mapped = series.map(mapping)
                    series = mapped.where(mapped.notna(), series)

                elif atype == "fillna":
                    value = action.get("value", "")
                    series = series.replace("", pd.NA)
                    series = series.fillna(value)

                elif atype == "to_number":
                    series = _to_number_series(series)

                elif atype == "currency_to_number":
                    series = _currency_to_number_series(series)

                elif atype == "round":
                    nd = action.get("ndigits", 2)
                    series = pd.to_numeric(series, errors="coerce").round(nd)

                elif atype in ("multiply", "divide", "add", "subtract"):
                    factor = action.get("value", 1)
                    series = pd.to_numeric(series, errors="coerce")
                    if atype == "multiply":
                        series = series * factor
                    elif atype == "divide":
                        series = series / factor
                    elif atype == "add":
                        series = series + factor
                    elif atype == "subtract":
                        series = series - factor

                else:
                    logging.warning(f"Unknown transform '{atype}' on column '{col}' — skipped")

            df[col] = series

    return df

def _match_any_regex(name: str, patterns: List[str]) -> bool:
    for pat in patterns or []:
        try:
            if re.search(pat, name):
                return True
        except re.error as e:

def filter_sheets(file_path: str, sheets: Dict[str, pd.DataFrame], cfg: Dict[str, Any]) -> Dict[str, pd.DataFrame]:
    rules = cfg.get("sheet_filters") or {}
    inc = set(rules.get("include", []) or [])
    exc = set(rules.get("exclude", []) or [])
    inc_rx = rules.get("include_regex", []) or []
    exc_rx = rules.get("exclude_regex", []) or []

    if not (inc or exc or inc_rx or exc_rx):
        return sheets

    out: Dict[str, pd.DataFrame] = {}
    for sname, df in sheets.items():
        name = str(sname)
        include_ok = True
        if inc or inc_rx:
            include_ok = (name in inc) or _match_any_regex(name, inc_rx)
        if include_ok and (name in exc or _match_any_regex(name, exc_rx)):
            include_ok = False
        if include_ok:
            out[name] = df

    logging.info(f"Sheet filter on '{os.path.basename(file_path)}': kept {list(out.keys()) or 'none'}")
    return out

def apply_type_hints(df: pd.DataFrame, cfg: Dict[str, Any]) -> pd.DataFrame:
    th = cfg.get("type_hints") or {}
    if not th.get("enabled", False):
        return df
    cols_map: Dict[str, str] = th.get("columns") or {}
    errors = th.get("errors", "coerce")
    df = df.copy()

    for col, t in cols_map.items():
        if col not in df.columns:
            continue
        t = (str(t) or "").lower()
        try:
            if t in ("number", "float", "double"):
                df[col] = pd.to_numeric(df[col], errors=errors)
            elif t in ("integer", "int"):
                df[col] = pd.to_numeric(df[col], errors=errors).astype("Int64")
            elif t in ("string", "text"):
                df[col] = df[col].astype("string")
            elif t in ("boolean", "bool"):
                s = df[col].astype(str).str.strip().str.lower()
                map_bool = {"true": True, "t": True, "1": True, "yes": True, "y": True,
                            "false": False, "f": False, "0": False, "no": False, "n": False, "": pd.NA, "nan": pd.NA}
                df[col] = s.map(map_bool).astype("boolean")
            elif t in ("category", "categorical"):
                df[col] = df[col].astype("category")
            else:
                logging.warning(f"type_hints: unsupported type '{t}' for column '{col}', skipped")
        except Exception as e:
            logging.warning(f"type_hints failed on '{col}' as '{t}': {e}")

    return df

def _apply_rule(df: pd.DataFrame, rule: Dict[str, Any]) -> pd.Series:
    col = rule.get("column")
    op = (rule.get("op") or "").lower()
    val = rule.get("value", None)

    if col not in df.columns:
        logging.warning(f"Filter rule skipped: column '{col}' not in DataFrame")
        return pd.Series([True] * len(df), index=df.index)

    s = df[col]

    if op == "notnull":
        return s.notna()
    if op == "isnull":
        return s.isna()

    if op in (">", ">=", "<", "<=", "==", "!="):
        try:
            return eval(f"s {op} @val")
        except Exception:
            try:
                sn = pd.to_numeric(s, errors="coerce")
                return eval(f"sn {op} @val")
            except Exception as e:
                logging.warning(f"Rule '{col} {op} {val}' failed: {e}")
                return pd.Series([True] * len(df), index=df.index)

    if op == "contains":
        return s.astype(str).str.contains(str(val), na=False, regex=True)
    if op == "in":
        return s.isin(val if isinstance(val, list) else [val])
    if op == "notin":
        return ~s.isin(val if isinstance(val, list) else [val])

    logging.warning(f"Unknown filter op '{op}' in rule on '{col}'")
    return pd.Series([True] * len(df), index=df.index)


def apply_filter_spec(df: pd.DataFrame, spec: Dict[str, Any]) -> pd.DataFrame:
    if not spec:
        return df
    out = df.copy()

    for q in spec.get("queries", []) or []:
        try:
            before = len(out)
            out = out.query(q, engine="python")
            logging.info(f"[item-filter] query '{q}': {before} -> {len(out)}")
        except Exception as e:
            logging.warning(f"[item-filter] query failed '{q}': {e}")

    rules: List[Dict[str, Any]] = spec.get("rules") or []
    if rules:
        mask = pd.Series([True] * len(out), index=out.index)
        for r in rules:
            try:
                mask &= _apply_rule(out, r)
            except Exception as e:
                logging.warning(f"[item-filter] rule failed {r}: {e}")
        before = len(out)
        out = out[mask]
        logging.info(f"[item-filter] rules: {before} -> {len(out)}")

    dna = spec.get("drop_na") or {}
    cols = dna.get("columns") or []
    if cols:
        how = dna.get("how", "any")
        before = len(out)
        out = out.dropna(subset=[c for c in cols if c in out.columns], how=how)
        logging.info(f"[item-filter] drop_na({cols}, how={how}): {before} -> {len(out)}")

    return out

def _ensure_date_bucket(df: pd.DataFrame, item: Dict[str, Any], cfg: Dict[str, Any]) -> pd.DataFrame:
    db = item.get("date_bucket")
    if not db:
        return df
    col = db.get("column")
    if not col or col not in df.columns:
        return df
    freq = db.get("freq", "MS")
    label = db.get("label", "bucket")
    parse_if_needed = bool(db.get("parse_if_needed", True))

    series = df[col]
    if not pd.api.types.is_datetime64_any_dtype(series):
        if parse_if_needed:
            dp = cfg.get("date_parsing") or {}
            dayfirst = bool(dp.get("dayfirst", False))
            series = pd.to_datetime(series, errors="coerce", dayfirst=dayfirst, infer_datetime_format=True)
        else:
            return df

    df = df.copy()
    df[label] = series.dt.to_period(freq).dt.to_timestamp()
    return df


def _append_totals_row(df: pd.DataFrame, label: str = "TOTAL", label_column: Optional[str] = None, funcs: Optional[List[str]] = None) -> pd.DataFrame:
    if df.empty:
        return df
    funcs = funcs or ["sum"]
    nums = df.select_dtypes(include="number")

    # compute totals with the first aggregating func that works (default sum)
    if not nums.empty:
        if "sum" in funcs:
            totals = nums.sum(numeric_only=True)
        elif "mean" in funcs:
            totals = nums.mean(numeric_only=True)
        elif "max" in funcs:
            totals = nums.max(numeric_only=True)
        elif "min" in funcs:
            totals = nums.min(numeric_only=True)
        else:
            totals = nums.sum(numeric_only=True)
    else:
        totals = pd.Series(dtype="float64")

    # build totals row with all columns
    total_row = {c: pd.NA for c in df.columns}
    for c in totals.index:
        if c in total_row:
            total_row[c] = totals[c]

    # place label
    if label_column and label_column in df.columns:
        total_row[label_column] = label
    else:
        # put label in the first non-numeric column if available
        nonnums = [c for c in df.columns if c not in nums.columns]
        if nonnums:
            total_row[nonnums[0]] = label

    return pd.concat([df, pd.DataFrame([total_row])], ignore_index=True)


def build_grouping_summaries(df: pd.DataFrame, cfg: Dict[str, Any]) -> Dict[str, pd.DataFrame]:
    gs = cfg.get("grouping_summaries") or {}
    if not gs.get("enabled", False):
        return {}
    out: Dict[str, pd.DataFrame] = {}
    items = gs.get("items") or []

    for item in items:
        name = item.get("name") or "Summary"
        work = df.copy()

        # Optional date bucket
        work = _ensure_date_bucket(work, item, cfg)

        # Optional per-item filter (before grouping)
        work = apply_filter_spec(work, item.get("filter") or {})

        # Pivot path?
        pivot = item.get("pivot")
        if pivot:
            rows = pivot.get("rows") or []
            cols = pivot.get("cols") or []
            values = pivot.get("values") or {}
            if not rows or not cols or not values:
                logging.warning(f"pivot '{name}': needs rows, cols, values")
                continue

            # build agg dict for pivot_table
            agg_dict: Dict[str, List[str]] = {}
            for vcol, funcs in values.items():
                if isinstance(funcs, str):
                    agg_dict[vcol] = [funcs]
                elif isinstance(funcs, list):
                    agg_dict[vcol] = funcs
                else:
                    logging.warning(f"pivot '{name}': invalid funcs for {vcol}")
            try:
                pt = pd.pivot_table(
                    work,
                    index=[r for r in rows if r in work.columns],
                    columns=[c for c in cols if c in work.columns],
                    values=[v for v in agg_dict.keys() if v in work.columns],
                    aggfunc={k: v for k, v in agg_dict.items()},
                    fill_value=pivot.get("fill_value", None),
                    margins=bool(pivot.get("margins", False)),
                    margins_name=pivot.get("margins_name", "All"),
                    dropna=False,
                )
                # Flatten MultiIndex columns
                if isinstance(pt.columns, pd.MultiIndex):
                    pt.columns = ["_".join([str(x) for x in tup if str(x) != ""]) for tup in pt.columns.to_flat_index()]
                pt = pt.reset_index()

                # Optional totals row (usually not needed when margins=true)
                tr = item.get("totals_row") or {}
                if tr.get("enabled", False):
                    pt = _append_totals_row(
                        pt,
                        label=tr.get("label", "TOTAL"),
                        label_column=tr.get("label_column"),
                        funcs=tr.get("funcs", ["sum"])
                    )

                out[name] = pt
                logging.info(f"Built pivot summary '{name}' with {len(pt)} rows")
            except Exception as e:
                logging.warning(f"pivot '{name}' failed: {e}")
            continue  # pivot handled, go next item

        # Standard group-by path
        group_by = item.get("group_by") or []
        aggs = item.get("aggregations") or {}
        sort_by = item.get("sort_by") or []

        if not group_by or not aggs:
            logging.warning(f"grouping item '{name}': requires group_by and aggregations or a pivot block")
            continue

        agg_dict: Dict[str, List[str]] = {}
        for col, funcs in aggs.items():
            if isinstance(funcs, str):
                agg_dict[col] = [funcs]
            elif isinstance(funcs, list):
                agg_dict[col] = funcs
            else:
                logging.warning(f"Invalid aggregation spec for '{col}' in '{name}'")

        try:
            summary = work.groupby([g for g in group_by if g in work.columns]).agg(agg_dict)
            summary.columns = ["_".join([c for c in tup if c]) for tup in summary.columns.to_flat_index()]
            summary = summary.reset_index()
            if sort_by:
                sort_cols = [c for c in sort_by if c in summary.columns]
                if sort_cols:
                    summary = summary.sort_values(by=sort_cols)

            # Optional totals row
            tr = item.get("totals_row") or {}
            if tr.get("enabled", False):
                summary = _append_totals_row(
                    summary,
                    label=tr.get("label", "TOTAL"),
                    label_column=tr.get("label_column"),
                    funcs=tr.get("funcs", ["sum"])
                )

            out[name] = summary
            logging.info(f"Built grouping summary '{name}' with {len(summary)} rows")
        except Exception as e:
            logging.warning(f"grouping item '{name}' failed: {e}")

    return out

def apply_computed_columns(df: pd.DataFrame, cfg: Dict[str, Any]) -> pd.DataFrame:
    cc = cfg.get("computed_columns") or {}
    exprs = cc.get("expressions") or []
    if not exprs:
        return df
    df = df.copy()
    for expr in exprs:
        try:
            if "=" in expr:
                df = df.eval(expr, engine="python")
            else:
                df.eval(expr, engine="python", inplace=True)
            logging.info(f"Computed: {expr}")
        except Exception as e:
            logging.warning(f"Failed computed expression '{expr}': {e}")
    return df


def apply_global_row_filters(df: pd.DataFrame, cfg: Dict[str, Any]) -> pd.DataFrame:
    gf = cfg.get("global_row_filters") or {}
    queries: List[str] = gf.get("queries") or []
    rules: List[Dict[str, Any]] = gf.get("rules") or []
    dropna = gf.get("drop_na") or {}

    out = df.copy()

    for q in queries:
        try:
            before = len(out)
            out = out.query(q, engine="python")
            logging.info(f"Filter query '{q}': {before} -> {len(out)}")
        except Exception as e:
            logging.warning(f"Query '{q}' failed: {e}")

    if rules:
        mask = pd.Series([True] * len(out), index=out.index)
        for r in rules:
            try:
                mask &= _apply_rule(out, r)
            except Exception as e:
                logging.warning(f"Rule failed {r}: {e}")
        before = len(out)
        out = out[mask]
        logging.info(f"Filter rules applied: {before} -> {len(out)}")

    cols = dropna.get("columns") or []
    if cols:
        how = dropna.get("how", "any")
        before = len(out)
        out = out.dropna(subset=[c for c in cols if c in out.columns], how=how)
        logging.info(f"drop_na({cols}, how={how}): {before} -> {len(out)}")

    return out

def unify_columns(dfs: List[pd.DataFrame]) -> List[pd.DataFrame]:
    all_cols = set()
    for df in dfs:
        all_cols.update(map(str, df.columns))
    all_cols = list(all_cols)
    unified = []
    for df in dfs:
        df.columns = list(map(str, df.columns))
        unified.append(df.reindex(columns=all_cols))
    return unified


def merge_files(
    cfg: Dict[str, Any],
    input_dir: str,
    include_patterns: List[str],
    sheet_name: Optional[str],
    header_row: Optional[int],
    add_source_columns: bool,
    deduplicate: bool,
    dedup_key_columns: List[str],
    csv_delimiter: str,
    encoding_pref: str,
) -> (pd.DataFrame, pd.DataFrame, Dict[str, pd.DataFrame]):
    records = []
    summary_rows = []

    for pattern in include_patterns:
        for path in glob.glob(os.path.join(input_dir, pattern)):
            if os.path.isdir(path):
                continue
            ext = os.path.splitext(path)[1].lower()

            if ext in [".csv", ".tsv"]:
                delim = csv_delimiter if ext == ".csv" else "\t"
                try:
                    df = read_csv_safely(path, delimiter=delim, encoding_pref=encoding_pref)
                    if add_source_columns:
                        df["source_file"] = os.path.basename(path)
                        df["source_sheet"] = ""
                    df = normalize_and_map_df_columns(df, cfg)
                    df = parse_dates_inplace(df, cfg)
                    df = apply_value_transforms(df, cfg)
                    df = apply_type_hints(df, cfg)

                    records.append(df)
                    summary_rows.append({"source_file": os.path.basename(path), "source_sheet": "", "rows": len(df)})
                    logging.info(f"Loaded CSV: {os.path.basename(path)} rows={len(df)}")
                except Exception as e:
                    logging.error(f"Error reading CSV '{path}': {e}\n{traceback.format_exc()}")

            elif ext in [".xlsx", ".xls"]:
                sheets = read_excel_sheets(path, sheet_name, header_row)
                sheets = filter_sheets(path, sheets, cfg)
                for sname, df in sheets.items():
                    if not isinstance(df, pd.DataFrame):
                        continue
                    if add_source_columns:
                        df["source_file"] = os.path.basename(path)
                        df["source_sheet"] = sname
                    df = normalize_and_map_df_columns(df, cfg)
                    df = parse_dates_inplace(df, cfg)
                    df = apply_value_transforms(df, cfg)
                    df = apply_type_hints(df, cfg)

                    records.append(df)
                    summary_rows.append({"source_file": os.path.basename(path), "source_sheet": sname, "rows": len(df)})
                logging.info(f"Loaded Excel: {os.path.basename(path)} sheets={len(sheets)}")
            else:
                logging.info(f"Skipped unsupported file: {os.path.basename(path)}")

    if not records:
        logging.warning("No data files found to merge.")
        return pd.DataFrame(), pd.DataFrame(), {}

    unified = unify_columns(records)
    merged = pd.concat(unified, ignore_index=True)

    merged = enforce_final_column_order(merged, cfg)

    cc_when = (cfg.get("computed_columns") or {}).get("when", "before").lower()
    if cc_when == "before":
        merged = apply_computed_columns(merged, cfg)

    if deduplicate:
        if dedup_key_columns:
            before = len(merged)
            merged = merged.drop_duplicates(subset=dedup_key_columns)
            logging.info(f"Deduplicated by keys {dedup_key_columns}: {before} -> {len(merged)}")
        else:
            before = len(merged)
            merged = merged.drop_duplicates()
            logging.info(f"Deduplicated full rows: {before} -> {len(merged)}")

    merged = apply_global_row_filters(merged, cfg)

    if cc_when == "after":
        merged = apply_computed_columns(merged, cfg)

    group_sheets = build_grouping_summaries(merged, cfg)

    summary = pd.DataFrame(summary_rows)
    if not summary.empty:
        summary = summary.groupby(["source_file", "source_sheet"], as_index=False)["rows"].sum()
        summary["total_rows_after_merge"] = len(merged)

    return merged, summary, group_sheets
```

```python
def _apply_rule(df: pd.DataFrame, rule: Dict[str, Any]) -> pd.Series:
    col = rule.get("column")
    op = (rule.get("op") or "").lower()
    val = rule.get("value", None)

    if col not in df.columns:
        logging.warning(f"Filter rule skipped: column '{col}' not in DataFrame")
        return pd.Series([True] * len(df), index=df.index)

    s = df[col]

    if op == "notnull":
        return s.notna()
    if op == "isnull":
        return s.isna()

    if op in (">", ">=", "<", "<=", "==", "!="):
        try:
            return eval(f"s {op} @val")
        except Exception:
            try:
                sn = pd.to_numeric(s, errors="coerce")
                return eval(f"sn {op} @val")
            except Exception as e:
                logging.warning(f"Rule '{col} {op} {val}' failed: {e}")
                return pd.Series([True] * len(df), index=df.index)

    if op == "contains":
        return s.astype(str).str.contains(str(val), na=False, regex=True)
    if op == "in":
        return s.isin(val if isinstance(val, list) else [val])
    if op == "notin":
        return ~s.isin(val if isinstance(val, list) else [val])

    logging.warning(f"Unknown filter op '{op}' in rule on '{col}'")
    return pd.Series([True] * len(df), index=df.index)

def _percent_of_total(df: pd.DataFrame, spec: Dict[str, Any]) -> pd.DataFrame:
    if not spec or not spec.get("enabled", False) or df.empty:
        return df
    cols = spec.get("columns") or []
    # Auto-detect numeric cols if none specified
    if not cols:
        cols = [c for c in df.columns if pd.api.types.is_numeric_dtype(df[c])]
    by = spec.get("by") or []
    suffix = spec.get("suffix", "_pct")
    scale = spec.get("scale", 100)
    decimals = int(spec.get("decimals", 2))

    out = df.copy()
    if by:
        group = out.groupby([b for b in by if b in out.columns], dropna=False)
        for col in cols:
            if col not in out.columns:
                continue
            totals = group[col].transform("sum")
            pct = (out[col] / totals.replace({0: pd.NA})) * scale
            out[f"{col}{suffix}"] = pct.round(decimals)
    else:
        for col in cols:
            if col not in out.columns:
                continue
            total = out[col].sum()
            denom = total if total not in (0, None) else pd.NA
            pct = (out[col] / denom) * scale
            out[f"{col}{suffix}"] = pct.round(decimals)
    return out


def _apply_topn(df: pd.DataFrame, spec: Dict[str, Any]) -> pd.DataFrame:
    if not spec or not spec.get("enabled", False) or df.empty:
        return df
    sort_col = spec.get("sort_column")
    if not sort_col or sort_col not in df.columns:
        logging.warning("topn: sort_column missing or not found; skipped")
        return df
    n = int(spec.get("n", 5))
    asc = bool(spec.get("ascending", False))
    by = spec.get("by") or []
    keep_ties = bool(spec.get("keep_ties", True))

    work = df.sort_values(by=sort_col, ascending=asc)
    if not by:
        if keep_ties and n < len(work):
            cutoff = work.iloc[n - 1][sort_col]
            mask = work[sort_col] >= cutoff if not asc else work[sort_col] <= cutoff
            return work[mask]
        return work.head(n)
    else:
        # group-wise head with ties preservation
        def _head_with_ties(g):
            if n >= len(g):
                return g
            g_sorted = g  # already sorted by outer sort
            cutoff = g_sorted.iloc[n - 1][sort_col]
            mask = g_sorted[sort_col] >= cutoff if not asc else g_sorted[sort_col] <= cutoff
            return g_sorted[mask] if keep_ties else g_sorted.head(n)

        return work.groupby([b for b in by if b in work.columns], group_keys=False, dropna=False).apply(_head_with_ties)


def _append_totals_row(df: pd.DataFrame, label: str = "TOTAL", label_column: Optional[str] = None, funcs: Optional[List[str]] = None) -> pd.DataFrame:
    if df.empty:
        return df
    funcs = funcs or ["sum"]
    # Exclude % columns from totals
    numeric_cols = [c for c in df.select_dtypes(include="number").columns if not re.search(r"_pct$", c)]
    if not numeric_cols:
        return df

    if "sum" in funcs:
        totals = df[numeric_cols].sum(numeric_only=True)
    elif "mean" in funcs:
        totals = df[numeric_cols].mean(numeric_only=True)
    elif "max" in funcs:
        totals = df[numeric_cols].max(numeric_only=True)
    elif "min" in funcs:
        totals = df[numeric_cols].min(numeric_only=True)
    else:
        totals = df[numeric_cols].sum(numeric_only=True)

    total_row = {c: pd.NA for c in df.columns}
    for c in totals.index:
        total_row[c] = totals[c]

    if label_column and label_column in df.columns:
        total_row[label_column] = label
    else:
        nonnums = [c for c in df.columns if c not in numeric_cols]
        if nonnums:
            total_row[nonnums[0]] = label

    return pd.concat([df, pd.DataFrame([total_row])], ignore_index=True)

def _ensure_date_bucket(df: pd.DataFrame, item: Dict[str, Any], cfg: Dict[str, Any]) -> pd.DataFrame:
    db = item.get("date_bucket")
    if not db:
        return df
    col = db.get("column")
    if not col or col not in df.columns:
        return df
    freq = db.get("freq", "MS")
    label = db.get("label", "bucket")
    parse_if_needed = bool(db.get("parse_if_needed", True))

    series = df[col]
    if not pd.api.types.is_datetime64_any_dtype(series):
        if parse_if_needed:
            dp = cfg.get("date_parsing") or {}
            dayfirst = bool(dp.get("dayfirst", False))
            series = pd.to_datetime(series, errors="coerce", dayfirst=dayfirst, infer_datetime_format=True)
        else:
            return df

    df = df.copy()
    df[label] = series.dt.to_period(freq).dt.to_timestamp()
    return df


def build_grouping_summaries(df: pd.DataFrame, cfg: Dict[str, Any]) -> Dict[str, pd.DataFrame]:
    gs = cfg.get("grouping_summaries") or {}
    if not gs.get("enabled", False):
        return {}
    out: Dict[str, pd.DataFrame] = {}
    items = gs.get("items") or []

    for item in items:
        name = item.get("name") or "Summary"
        work = df.copy()

        # Date bucket (if any)
        work = _ensure_date_bucket(work, item, cfg)
        # Per-summary filter
        work = apply_filter_spec(work, item.get("filter") or {})

        pivot = item.get("pivot")
        if pivot:
            rows = pivot.get("rows") or []
            cols = pivot.get("cols") or []
            values = pivot.get("values") or {}
            if not rows or not cols or not values:
                logging.warning(f"pivot '{name}': needs rows, cols, values")
                continue

            agg_dict: Dict[str, List[str]] = {}
            for vcol, funcs in values.items():
                if isinstance(funcs, str):
                    agg_dict[vcol] = [funcs]
                elif isinstance(funcs, list):
                    agg_dict[vcol] = funcs
                else:
                    logging.warning(f"pivot '{name}': invalid funcs for {vcol}")

            try:
                pt = pd.pivot_table(
                    work,
                    index=[r for r in rows if r in work.columns],
                    columns=[c for c in cols if c in work.columns],
                    values=[v for v in agg_dict.keys() if v in work.columns],
                    aggfunc={k: v for k, v in agg_dict.items()},
                    fill_value=pivot.get("fill_value", None),
                    margins=bool(pivot.get("margins", False)),
                    margins_name=pivot.get("margins_name", "All"),
                    dropna=False,
                )
                if isinstance(pt.columns, pd.MultiIndex):
                    pt.columns = ["_".join([str(x) for x in tup if str(x) != ""]) for tup in pt.columns.to_flat_index()]
                pt = pt.reset_index()

                # % of total on pivot
                pt = _percent_of_total(pt, item.get("percent_of_total") or {})
                # Top-N (rare for pivot, but supported on flattened table)
                pt = _apply_topn(pt, item.get("topn") or {})

                # Totals row (usually not needed when margins=true)
                tr = item.get("totals_row") or {}
                if tr.get("enabled", False):
                    pt = _append_totals_row(
                        pt,
                        label=tr.get("label", "TOTAL"),
                        label_column=tr.get("label_column"),
                        funcs=tr.get("funcs", ["sum"])
                    )

                out[name] = pt
                logging.info(f"Built pivot summary '{name}' with {len(pt)} rows")
            except Exception as e:
                logging.warning(f"pivot '{name}' failed: {e}")
            continue

        # Standard group-by
        group_by = item.get("group_by") or []
        aggs = item.get("aggregations") or {}
        sort_by = item.get("sort_by") or []

        if not group_by or not aggs:
            logging.warning(f"grouping item '{name}': requires group_by and aggregations or a pivot block")
            continue

        agg_dict: Dict[str, List[str]] = {}
        for col, funcs in aggs.items():
            if isinstance(funcs, str):
                agg_dict[col] = [funcs]
            elif isinstance(funcs, list):
                agg_dict[col] = funcs
            else:
                logging.warning(f"Invalid aggregation spec for '{col}' in '{name}'")

        try:
            summary = work.groupby([g for g in group_by if g in work.columns]).agg(agg_dict)
            summary.columns = ["_".join([c for c in tup if c]) for tup in summary.columns.to_flat_index()]
            summary = summary.reset_index()

            if sort_by:
                sort_cols = [c for c in sort_by if c in summary.columns]
                if sort_cols:
                    summary = summary.sort_values(by=sort_cols)

            # % of total
            summary = _percent_of_total(summary, item.get("percent_of_total") or {})
            # Top-N
            summary = _apply_topn(summary, item.get("topn") or {})

            # Totals row (after filters/topN so totals reflect the shown rows)
            tr = item.get("totals_row") or {}
            if tr.get("enabled", False):
                summary = _append_totals_row(
                    summary,
                    label=tr.get("label", "TOTAL"),
                    label_column=tr.get("label_column"),
                    funcs=tr.get("funcs", ["sum"])
                )

            out[name] = summary
            logging.info(f"Built grouping summary '{name}' with {len(summary)} rows")
        except Exception as e:
            logging.warning(f"grouping item '{name}' failed: {e}")

    return out

def apply_computed_columns(df: pd.DataFrame, cfg: Dict[str, Any]) -> pd.DataFrame:
    cc = cfg.get("computed_columns") or {}
    exprs = cc.get("expressions") or []
    if not exprs:
        return df
    df = df.copy()
    for expr in exprs:
        try:
            if "=" in expr:
                df = df.eval(expr, engine="python")
            else:
                df.eval(expr, engine="python", inplace=True)
            logging.info(f"Computed: {expr}")
        except Exception as e:
            logging.warning(f"Failed computed expression '{expr}': {e}")

def apply_global_row_filters(df: pd.DataFrame, cfg: Dict[str, Any]) -> pd.DataFrame:
    gf = cfg.get("global_row_filters") or {}
    queries: List[str] = gf.get("queries") or []
    rules: List[Dict[str, Any]] = gf.get("rules") or []
    dropna = gf.get("drop_na") or {}

    out = df.copy()

    for q in queries:
        try:
            before = len(out)
            out = out.query(q, engine="python")
            logging.info(f"Filter query '{q}': {before} -> {len(out)}")
        except Exception as e:
            logging.warning(f"Query '{q}' failed: {e}")

    if rules:
        mask = pd.Series([True] * len(out), index=out.index)
        for r in rules:
            try:
                mask &= _apply_rule(out, r)
            except Exception as e:
                logging.warning(f"Rule failed {r}: {e}")
        before = len(out)
        out = out[mask]
        logging.info(f"Filter rules applied: {before} -> {len(out)}")

    cols = dropna.get("columns") or []
    if cols:
        how = dropna.get("how", "any")
        before = len(out)
        out = out.dropna(subset=[c for c in cols if c in out.columns], how=how)
        logging.info(f"drop_na({cols}, how={how}): {before} -> {len(out)}")

    return out

def unify_columns(dfs: List[pd.DataFrame]) -> List[pd.DataFrame]:
    all_cols = set()
    for df in dfs:
        all_cols.update(map(str, df.columns))
    all_cols = list(all_cols)
    unified = []
    for df in dfs:
        df.columns = list(map(str, df.columns))
        unified.append(df.reindex(columns=all_cols))
    return unified

        return pd.DataFrame(), pd.DataFrame(), {}

    unified = unify_columns(records)
```

```python
def _percent_of_parent(df: pd.DataFrame, spec: Dict[str, Any]) -> pd.DataFrame:
    """
    Compute % of parent for hierarchical groups on an already-aggregated summary.

    spec:
      enabled: bool
      hierarchy: ordered list of key cols present in df (must match group_by order)
      columns: list of numeric aggregated cols to compute against (e.g., 'amount_sum')
      suffix: str, default '_pp'
      scale: int/float, default 100
      decimals: int, default 2
      levels: optional subset of hierarchy levels to compute for (exclude root)
    """
    if not spec or not spec.get("enabled", False) or df.empty:
        return df

    hierarchy = [h for h in (spec.get("hierarchy") or []) if h in df.columns]
    if len(hierarchy) < 2:
        return df  # need at least parent->child

    cols = [c for c in (spec.get("columns") or []) if c in df.columns]
    if not cols:
        # auto-detect numeric aggregated cols if none specified
        cols = [c for c in df.columns if pd.api.types.is_numeric_dtype(df[c])]

    suffix = spec.get("suffix", "_pp")
    scale = spec.get("scale", 100)
    decimals = int(spec.get("decimals", 2))
    allowed_levels = set(spec.get("levels") or hierarchy[1:])  # all except root by default

    out = df.copy()

    # For each level deeper than root, compute col / parent_total over parent keys
    for i in range(1, len(hierarchy)):
        level = hierarchy[i]
        if level not in allowed_levels:
            continue
        parent_keys = hierarchy[:i]          # parent grouping keys
        # transform parent totals once per column
        group_parent = out.groupby(parent_keys, dropna=False)
        for col in cols:
            if col not in out.columns:
                continue
            denom = group_parent[col].transform("sum").replace({0: pd.NA})
            pct = (out[col] / denom) * scale
            out[f"{col}{suffix}_{level}"] = pct.round(decimals)

    return out
```

```python
      2) computing parent totals along the hierarchy
      3) pivoting back and concatenating % columns alongside originals

    Supports multiple metrics/funcs (iterated). Returns a wide DF with
    original numeric columns + % columns suffixed with e.g. '_pp_col_region'.
    """
    # Normalize specs
    rows = [c for c in rows if c in work.columns]
    cols = [c for c in cols if c in work.columns]
    hierarchy_cols = [c for c in hierarchy_cols if c in cols]
    if not rows or not cols or not hierarchy_cols:
        return pd.pivot_table(
            work, index=rows, columns=cols, values=list(values_spec.keys()),
            aggfunc={k: v for k, v in values_spec.items()}, fill_value=fill_value, dropna=False
        ).reset_index()

    # Make a tidy long-form aggregate for each metric/func, then compute % and pivot back, concat horizontally
    wide_frames = []

    for metric, funcs in values_spec.items():
        if metric not in work.columns:
            continue
        funcs = funcs if isinstance(funcs, list) else [funcs]

        for func in funcs:
            # 1) Aggregate long-form at full detail: rows + full cols
            group_keys = rows + cols
            agg = work.groupby(group_keys, dropna=False)[metric].agg(func).reset_index()
            metric_agg_name = f"{metric}_{func}"

            # 2) Compute % of parent at each deeper level in hierarchy
            long_df = agg.copy()
            # Track % columns by level
            pct_cols_long = []

            for i in range(1, len(hierarchy_cols)):
                level = hierarchy_cols[i]
                parent_keys = rows + hierarchy_cols[:i]  # parent grouping: rows + higher column levels
                child_keys  = rows + hierarchy_cols[: i + 1]

                # Parent totals: sum over all deeper children at this level
                parent_totals = long_df.groupby(parent_keys, dropna=False)[metric].transform("sum").replace({0: pd.NA})
                pct = (long_df[metric] / parent_totals) * scale
                pct = pct.round(decimals)
                col_name = f"{metric_agg_name}{suffix}_{level}"
                long_df[col_name] = pct
                pct_cols_long.append(col_name)

            # 3) Pivot back to wide for the base metric_agg and each pct col separately, then concat
            # Base numeric
            base_piv = pd.pivot_table(
                long_df,
                index=rows,
                columns=cols,
                values=metric,
                aggfunc="sum",          # already aggregated; sum across identical keys is idempotent
                fill_value=fill_value,
                dropna=False,
            )
            base_piv = _flatten_multiindex_columns(base_piv).reset_index()
            # Rename base columns to metric_agg_name prefix
            base_cols = [c for c in base_piv.columns if c not in rows]
            base_piv = base_piv.rename(columns={c: f"{metric_agg_name}_{c}" for c in base_cols})

            # Percent columns
            pct_pivs = []
            for pct_col in pct_cols_long:
                pp = pd.pivot_table(
                    long_df,
                    index=rows,
                    columns=cols,
                    values=pct_col,
                    aggfunc="max",        # each cell unique; max/min/sum are equivalent after prior groupby
                    fill_value=fill_value,
                    dropna=False,
                )
                pp = _flatten_multiindex_columns(pp).reset_index()
                # Already has pct_col names, but include column coordinate in flattened name
                pcols = [c for c in pp.columns if c not in rows]
                pp = pp.rename(columns={c: f"{pct_col}_{c}" for c in pcols})
                pct_pivs.append(pp)

            # Merge all frames for this metric+func on row keys
            merged = base_piv
            for pp in pct_pivs:
                merged = pd.merge(merged, pp, on=rows, how="outer")
            wide_frames.append(merged)

    # Final: merge all metric/func blocks on row keys
    if not wide_frames:
        return pd.DataFrame(columns=rows)

    out = wide_frames[0]
    for block in wide_frames[1:]:
        out = pd.merge(out, block, on=rows, how="outer")

    return out
```
